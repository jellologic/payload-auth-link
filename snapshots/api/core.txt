/**
 * Inputs to the shared production-secure rule. Every provider's
 * cookie-name derivation funnels through {@link resolveSecureCookies}
 * with these three inputs so secure-cookie behavior cannot drift
 * between providers.
 */
export type SecureCookieRuleInput = {
    /** Explicit builder flag; wins over every other input. */
    readonly useSecureCookies?: boolean | undefined;
    /** Configured site origin; an https origin means secure cookies. */
    readonly baseURL?: string | undefined;
    /**
     * Better Auth object-form `baseURL` protocol (`'https'` / `'http'`),
     * which that SDK reads at runtime ahead of the string check. Providers
     * whose SDK has no object form leave this unset.

    readonly dynamicProtocol?: string | undefined;
};
/** Injectable environment for the production fallback below. */
export type SecureCookieEnvironment = {
    readonly nodeEnv?: string | undefined;
};
/**
 * One production-secure rule shared by every provider: an explicit
 * `useSecureCookies` wins, then the configured origin (https → secure),
 * then `NODE_ENV=production`. Pure and Edge-safe — no imports, no
 * globals: callers inject the environment (adapters read the live
 * `NODE_ENV`; an absent environment reads as non-production).
 *
 * Production deployments must set a canonical https `baseURL`: it is the
 * only input on which both SDKs and the session layer's origin anchoring
 * agree. Without it, production falls back to `NODE_ENV` detection, which
 * matches SDK behavior on https deployments but fails closed (Auth.js
 * session detection misses) on http ones.

export declare function resolveSecureCookies(input: SecureCookieRuleInput, environment?: SecureCookieEnvironment): boolean;

/**

 * only programmer errors and truly unexpected faults are thrown.
 *
 * Domain errors (invalid input at a boundary) stay distinct from application
 * errors (a valid operation that failed). Web adapters map them to status
 * codes; core never touches HTTP.
 */
/**
 * Base error: machine-readable `code`, human `message`, optional
 * `cause`. Leaf classes fix their code; adapters map codes to HTTP.
 */
export declare class AuthLinkError extends Error {
    readonly code: string;
    constructor(code: string, message: string, options?: {
        cause?: unknown;
    });
}

// ===== events.d.ts =====
import type { Email } from './email.js';
import type { ProviderId } from './ports.js';
/**
 * Provider-side identity lifecycle facts (CAP-14), translated by each
 * adapter from its native hooks (Better Auth `databaseHooks`, Auth.js
 * `events`) into this one core vocabulary. The plugin applies them
 * through `UserRepository`, so Payload stays the single user store
 * (invariant 1): an event can sync a verified email or unlink an
 * identity row, but it can never create or delete a Payload user.
 *
 * These are provider HOOKS observed in-process, not webhooks: nothing
 * here is delivered over the network, and the read-time sync in
 * `ensureLinkedUser` remains the backstop when an event is missed
 * (for example while the process was down).
 */
export type IdentityEvent = {
    /**
     * The provider changed the user's email. Applied to Payload
     * only when `verified` is true (D9: the provider owns the
     * email for verified changes only); an unverified change is
     * ignored and logged.
     */
    readonly type: 'email-changed';
    readonly providerId: ProviderId;
    readonly providerUserId: string;
    readonly email: Email;
    readonly verified: boolean;
} | {
    /**
     * The provider verified this email for the user. Payload users
     * carry no verification flag (verification gates linking, which
     * already happened), so applying means: sync the address if it
     * differs, otherwise a no-op.
     */
    readonly type: 'email-verified';
    readonly providerId: ProviderId;
    readonly providerUserId: string;
    readonly email: Email;
} | {
    /**
     * The provider account was unlinked from the provider user.
     * Applying removes the identity row from the Payload user; the
     * Payload user itself always remains (D9).
     */
    readonly type: 'identity-unlinked';
    readonly providerId: ProviderId;
    readonly providerUserId: string;
} | {
    /**
     * The provider user was deleted. Applying unlinks the identity
     * row; the Payload user remains, because Payload owns it (D9).
     */
    readonly type: 'user-deleted';
    readonly providerId: ProviderId;
    readonly providerUserId: string;
};
/**
 * Consumes one {@link IdentityEvent}. Adapters call every registered
 * handler from their provider hooks; handlers must not throw into the
 * provider operation (the plugin's applier logs failures instead, so
 * a Payload-side sync failure never aborts the provider-side change
 * that already committed — read-time sync is the backstop).
 */
export type IdentityEventHandler = (event: IdentityEvent) => Promise<void>;
/**
 * The fan-out behind an adapter's `onIdentityEvent`: adapters `emit`
 * narrowed events from their provider hooks; the plugin subscribes
 * once. Emission awaits every handler in subscription order.
 */
export type IdentityEventEmitter = {
    readonly emit: (event: IdentityEvent) => Promise<void>;
    readonly subscribe: (handler: IdentityEventHandler) => () => void;
};
/** Create an {@link IdentityEventEmitter} for an adapter. */
export declare function createIdentityEventEmitter(): IdentityEventEmitter;

import { type ResolvedSessionActor, type SessionAssurance } from './ports.js';
/**

 * UI adapters always send it with the value `1` on state-changing requests.
 */

/**
 * Default plugin session endpoint path. Payload serves custom endpoints
 * under `/api`; builders who customize `routes.api` override the endpoint
 * in the UI adapters instead of this constant.
 */
export declare const defaultSessionEndpointPath = "/api/auth-link/session";
/**
 * Injectible `fetch` shape. UI adapters default to the global `fetch`;
 * tests inject stubs. Same signature as `fetch` restricted to string URLs.
 */
export type HttpFetch = (input: string, init?: RequestInit) => Promise<Response>;
/**
 * Validate a configured origin. Returns it unchanged when it is an
 * absolute http(s) URL; throws `InvalidConfigError` otherwise. Call at
 * wiring time so a malformed base URL fails the build, never a request
 * (where `new URL` would throw a bare `TypeError` instead).
 */
export declare function normalizeBaseUrl(baseUrl: string): string;
/** Origin resolution: configured base URL plus the dev-only host fallback. */
export type OriginResolutionInput = {
    /** Validated configured origin; wins whenever set. */
    readonly baseUrl?: string | undefined;
    /** Request headers, for the host fallback only. */
    readonly headers: Headers;
    /**
     * Whether the request-host fallback is allowed. Host headers are
     * client-spoofable, so production session reads (which forward
     * cookies to the resolved origin) must pass `false`; local
     * development passes `true`.
     */
    readonly allowHostFallback: boolean;
};
/**
 * Single origin resolver behind every server-side session read
 * (`@payload-auth-link/web` verifier, Auth.js engine). The configured
 * base URL wins; without it, the request host is used only when
 * `allowHostFallback` is set. Throws `InvalidConfigError` when no
 * origin can be resolved — a wiring bug, fail loud rather than
 * fetching a nonsense URL.
 */
export declare function resolveOrigin(input: OriginResolutionInput): string;
/**
 * The session user: what the plugin's session endpoint returns inside its
 * `{ user }` envelope, and what the UI adapters hydrate from. Plain JSON —
 * roles are the wire form of `Role`, re-validated on parse.
 *
 * The PORT-4 fields are additive and optional, so payloads from older
 * servers (without them) still parse, and older clients ignore them.
 * Producers include a field only when it is known: `actor` only while
 * an operator impersonates the user, `organizationId` only when the
 * provider reports an active tenant. `assurance` is the exception that
 * proves the rule — when the provider reported it, it serializes even
 * when its values are `'unknown'`, because "the provider cannot tell"
 * is itself a fact. Consumers must treat absent or `'unknown'`
 * assurance as not satisfied (R3 fail-closed): it never proves a
 * method or MFA.
 */
export type SessionUser = {
    readonly id: string;
    readonly collection: string;
    readonly email: string;
    readonly roles: readonly string[];
    /** How the session was proven, when the provider reported it. */
    readonly assurance?: SessionAssurance | undefined;
    /**
     * The impersonating operator, present only while impersonating,
     * resolved against Payload users when linked (CAP-8b): the
     * `kind: 'payload-user'` form also carries the operator's Payload
     * user id; `kind: 'provider-only'` means no linked Payload user.
     */
    readonly actor?: ResolvedSessionActor | undefined;
    /** Provider-native active tenant, when one is active. */
    readonly organizationId?: string | undefined;
    /**
     * Present (`true`) only when a `requireMfaForRoles` policy (CAP-6)
     * withheld one or more of the user's stored roles from `roles`
     * because this session lacks MFA proof — the UI's signal to prompt
     * for a step-up sign-in instead of a flat "forbidden".
     */
    readonly mfaRequired?: true | undefined;
};
/**
 * Why {@link parseSessionUser} degraded an optional wire field instead
 * of trusting it (B20). Reported through
 * {@link ParseSessionUserOptions.onDegrade} so a producer/consumer
 * version skew is observable in logs instead of silent: an unknown
 * `AuthMethod` from a newer server lands here as
 * `'assurance-malformed'`, never as a mysteriously absent session.
 */
export type SessionUserDegradeReason = 'assurance-malformed' | 'actor-malformed' | 'organization-malformed';
/** Options for {@link parseSessionUser}. */
export type ParseSessionUserOptions = {
    /**
     * Called once per degraded optional field. Intended for a
     * `LoggerPort.warn` at the composition root; omitted by default,
     * which preserves the historical silent fail-closed behavior.
     */
    readonly onDegrade?: ((reason: SessionUserDegradeReason) => void) | undefined;
};
/**
 * Narrow unknown JSON to a `SessionUser`. This is the CLIENT wire
 * parser: core identity fields (`id`, `collection`, `email`,
 * `roles`) stay strict and fail closed (`null`) on any deviation,
 * because a partial identity is never a session.
 *
 * The optional PORT-4 fields degrade instead of nuking the user:
 * unknown keys inside `assurance`/`actor` are ignored (forward
 * compatibility with newer servers), and a present-but-malformed
 * optional degrades fail-closed-for-policy — `assurance` to
 * `'unknown'`/`'unknown'` (which never satisfies a policy requiring
 * proof), `actor`/`organizationId` to absent. Dropping a malformed
 * client-side actor marker is safe ONLY because server-side
 * authorization and audit never route through this parser: the
 * plugin extracts the actor from `req.user` with its own fail-closed
 * server extraction (a malformed actor there still denies). The
 * explicit `null` a producer may carry for "no actor / no tenant"
 * normalizes to absent; `null` assurance degrades to `'unknown'`.
 */
export declare function parseSessionUser(value: unknown, options?: ParseSessionUserOptions): SessionUser | null;
/**
 * The guest principal as it travels on the wire (CAP-10): the session
 * endpoint answers `{ user: null, guest }` for a live anonymous
 * provider session. A guest has NO Payload user, roles, or access —
 * `id` is only a stable key for anonymous app state.
 */
export type SessionGuest = {
    readonly id: string;
    readonly providerId: string;
    /** ISO instant the underlying provider session expires. */
    readonly expiresAt: string;
};
/** Narrow unknown JSON to a `SessionGuest`; `null` on any deviation. */
export declare function parseSessionGuest(value: unknown): SessionGuest | null;
/**
 * Parse a session-endpoint `Response` through its `{ guest }`
 * envelope (CAP-10). Fails closed exactly like
 * {@link parseSessionResponse}; note a guest body parses to `null`
 * there (its `user` is `null`), so user-only consumers stay correct
 * without knowing guests exist.
 */
export declare function parseGuestResponse(response: Pick<Response, 'ok' | 'json'>): Promise<SessionGuest | null>;
/**
 * Parse a session-endpoint `Response` through its `{ user }` envelope.
 * Fails closed (`null`) on non-2xx statuses, unreadable JSON, non-record
 * bodies, and malformed users — the single envelope parser shared by
 * the client reader (`react`) and the server verifier (`web`).
 */
export declare function parseSessionResponse(response: Pick<Response, 'ok' | 'json'>): Promise<SessionUser | null>;

// ===== linking.d.ts =====
import { type Result } from 'neverthrow';
import { z } from 'zod';
import { InvalidConfigError } from './errors.js';
/**
 * Declared policy for linking a provider identity to a Payload user
 * (PORT-3). The policy names which verified identifier claims may act
 * as the linking authority in `ensureLinkedUser`:
 *
 * - `'verified-email'` (the default): only a provider-verified email
 *   claim links to an existing user or provisions a new one. This is
 *   the pre-PORT-3 hard-coded rule, unchanged.
 * - `'verified-email-or-phone'`: the email rule above, plus one
 *   additional branch for phone-only identities (no email claim): a
 *   provider-verified E.164 phone claim links to the _existing_ user
 *   resolved through `UserRepository.findByPhone`. The branch never
 *   provisions a user — an unmatched verified phone refuses with
 *   `UnverifiedIdentityError`.
 *
 * Anonymous identities (`identity.anonymous === true`) never link
 * under any policy (D3).
 */
/**
 * Linking-policy boundary schema. Strict and discriminated on `kind`:
 * unknown kinds and unknown keys are rejected, never silently mapped
 * to the default. Defined before {@link LinkingPolicy} so the domain
 * type derives from the schema (single source, B16): the hand-written
 * `readonly` union it replaced was assignable but not identical to
 * the schema output — exactly the drift the type-alarms pin catches.
 */
export declare const linkingPolicySchema: z.ZodDiscriminatedUnion<[z.ZodObject<{
    kind: z.ZodLiteral<"verified-email">;
}, z.core.$strict>, z.ZodObject<{
    kind: z.ZodLiteral<"verified-email-or-phone">;
}, z.core.$strict>], "kind">;
export type LinkingPolicy = Readonly<z.output<typeof linkingPolicySchema>>;
/**
 * The default linking policy: verified email only. Used by
 * `ensureLinkedUser` when `EnsureLinkedUserInput.linkingPolicy` is
 * omitted, so existing callers keep byte-identical behavior.
 */
export declare const defaultLinkingPolicy: LinkingPolicy;
/**
 * Parse an untyped linking policy (for example from plugin options).
 * Returns {@link InvalidConfigError} on any shape the strict schema
 * rejects — wiring fails fast instead of degrading to the default.
 */
export declare function parseLinkingPolicy(value: unknown): Result<LinkingPolicy, InvalidConfigError>;

 * `lastLoginMethod` must never back per-session assurance).

/**
 * A {@link SessionActor} resolved against Payload users (CAP-8b), as
 * carried on the plugin-level {@link Session} and the `SessionUser`
 * wire shape. Discriminated so an unresolved operator can never be
 * mistaken for a Payload user:
 *
 * - `payload-user`: the operator's provider identity is linked to a
 *   Payload user; `userId` is that user's Payload document id.
 * - `provider-only`: the operator has no linked Payload user (e.g. a
 *   provider-side admin who has never signed in through Payload).
 *   Fail-closed by design: the session still verifies and the actor
 *   is still audit-logged and still restricted by impersonation
 *   access policy, but the actor stays in explicit provider-id form

 *   as soon as the operator links a Payload user.
 */
export type ResolvedSessionActor = {
    readonly kind: 'payload-user';
    readonly providerUserId: string;
    readonly userId: UserId;

    readonly kind: 'provider-only';
    readonly providerUserId: string;
};

/**
 * One identity provider behind the port. Register via the registry.
 *
 * Providers own their OAuth dance (their SDK mounts its own routes and
 * persists via its database adapter); the plugin only needs to verify and
 * revoke provider sessions and bridge them to Payload users.
 */
export interface IdentityProviderPort {
    readonly id: ProviderId;
    /**
     * Cookie names carrying this provider's session (e.g. Better Auth's
     * `better-auth.session_token`). Surfaces use them for middleware
     * fast-paths and for clearing stale provider cookies on logout — the
     * provider itself remains the only reader of their values.
     */
    readonly sessionCookieNames: readonly string[];
    /**
     * Clearing attributes for one session cookie on logout: `secure`
     * (required to overwrite `Secure` cookies, which prefix inference
     * misses when the name is customized), `domain`, `path`, `sameSite`.
     * Optional for backward compatibility — providers that omit it fall
     * back to prefix-derived attributes (see `expiringCookieOptions`).
     */
    readonly sessionCookieClearOptions?: ((name: string) => CookieOptions | undefined) | undefined;
    /**
     * Extra provider cookies to expire on logout that do NOT themselves
     * indicate a live session, so they must stay out of
     * `sessionCookieNames` (middleware fast-paths read that list as
     * "signed in"). Example: Better Auth's `admin_session`, which holds
     * the impersonating operator's session token — leaving it behind on
     * logout would keep a resumable operator credential in the browser
     * after the user believes they signed out. Optional: providers
     * without such cookies omit it.
     */
    readonly logoutCookieNames?: readonly string[] | undefined;
    /**
     * The capabilities this adapter honestly supports today (PORT-2).
     * Declared, never discovered: a plugin option or policy that needs
     * a capability absent here fails at wiring with
     * `UnsupportedCapabilityError` instead of silently degrading. An
     * adapter declares a capability only once it maps the provider's
     * native mechanism to the core claims/assurance vocabulary.
     */
    readonly capabilities: ReadonlySet<Capability>;
    getSession(headers: Headers): ResultAsync<ProviderSession | null, ProviderError>;
    revokeSession(sessionId: string): ResultAsync<void, ProviderError>;
    /**
     * Revoke every live session of one provider user (CAP-7 suspend).
     * Optional: only adapters that declare `'revoke-user-sessions'` in
     * {@link IdentityProviderPort.capabilities} implement it; providers
     * on stateless JWT sessions honestly omit it (the Payload-side
     * suspended check still refuses at verify time).
     */
    readonly revokeUserSessions?: ((providerUserId: string) => ResultAsync<void, ProviderError>) | undefined;
    /**
     * Mint a provider session for one provider user — the supported
     * server-side path behind `authLinkServer.createSessionFor` (issue
     * #54). Optional: only adapters that declare
     * `'server-session-create'` implement it; the server API refuses
     * session creation with a typed error when the configured provider
     * cannot (e.g. stateless Auth.js).
     */
    readonly createProviderSession?: ((providerUserId: string, options: {
        readonly expiresAt: Date;
    }) => ResultAsync<{
        readonly sessionId: string;
        readonly token: string;
    }, ProviderError>) | undefined;
    /**
     * Enumerate one provider user's live sessions — the read side of
     * `authLinkServer.listSessions`. Optional: only adapters declaring
     * `'server-session-list'` implement it.
     */
    readonly listProviderSessions?: ((providerUserId: string) => ResultAsync<readonly {
        readonly sessionId: string;
        readonly expiresAt: Date | null;
    }[], ProviderError>) | undefined;
    /**
     * Verify an API key and resolve its owner + metadata —
     * `authLinkServer.verifyApiKey`. Optional: only adapters declaring
     * `'api-key-verify'` implement it.
     */
    readonly verifyApiKey?: ((key: string) => ResultAsync<ApiKeyVerification | null, ProviderError>) | undefined;
    /**
     * The provider's correctly signed session cookie for one session
     * token — `authLinkServer.sessionCookie`. Optional: providers whose
     * session transport is not a cookie (stateless Auth.js) omit it.
     */
    readonly sessionCookie?: ((token: string) => string) | undefined;
    /**
     * Enumerate every session id presented by this request beyond the
     * active session (CAP-13b multi-session logout): the ids are
     * provider-verified handles suitable for {@link revokeSession},
     * never parsed out of cookie names. Optional: only multi-session
     * capable adapters implement it; logout revokes each returned id
     * and expires every presented multi-session cookie.
     */
    readonly listPresentedSessionIds?: ((headers: Headers) => ResultAsync<readonly string[], ProviderError>) | undefined;
    /**
     * Names of presented cookies that must be expired on logout in
     * addition to `sessionCookieNames`/`logoutCookieNames`, computed
     * from the actual request (CAP-13b: multi-session cookie names are
     * per-token dynamic, so no static list can contain them). Optional.
     */
    readonly presentedLogoutCookieNames?: ((headers: Headers) => readonly string[]) | undefined;
    /**
     * Subscribe to this provider's identity lifecycle events (CAP-14),
     * translated from its native hooks. Returns an unsubscribe
     * function. Optional: adapters whose provider exposes no usable
     * hooks omit it and rely on read-time sync. The plugin subscribes
     * at `onInit` (the composition root) and applies events through
     * `UserRepository`; builders never wire this by hand.
     */
    readonly onIdentityEvent?: ((handler: IdentityEventHandler) => () => void) | undefined;
}

// ===== redirect.d.ts =====
/**
 * Post-login redirect safety. The sign-in page receives its destination
 * as a query param, and query params are attacker-controlled input:
 * blindly pushing them is an open redirect. Every consumer of that
 * param (sign-in forms, `SignOutButton`, adapters) funnels through here.
 * Lives in core so UI packages can use it without a web dependency;
 * `@payload-auth-link/web` re-exports it unchanged.
 */
/** Destination policy for {@link safeRedirectPath}. */
export type SafeRedirectOptions = {
    /**
     * Sign-in page path. A destination equal to it (with or without a
     * query) falls back: landing a fresh login on the sign-in page loops.
     * Defaults to `/sign-in`, matching `decideAuthRedirect`. `null`
     * disables the rule for destinations where sign-in is legitimate
     * (post-sign-out navigation).
     */
    readonly signInPath?: string | null | undefined;
    /**
     * API mount prefix. Destinations under it fall back: a browser
     * navigation to a JSON or mutating endpoint is never a page to land
     * on. Defaults to `/api` (Payload's default `routes.api`).
     */
    readonly apiPrefix?: string | undefined;
};
/**
 * Narrow a post-login destination to a safe in-app page path. Returns
 * `value` only when it is a relative path (single leading slash, no
 * scheme, no backslashes, no control characters) that is neither the
 * sign-in page nor under the API prefix; anything else yields
 * `fallback`. Absolute URLs, protocol-relative URLs (`//evil`), and
 * values with surrounding whitespace (header/parsing differentials) all
 * fall back.
 */
export declare function safeRedirectPath(value: string | null | undefined, fallback?: string, options?: SafeRedirectOptions): string;

// ===== testing/provider-conformance.d.ts =====
import type { AuthMethod, IdentityProviderPort, SessionActor, SessionChannel } from '../ports.js';
/** One named conformance check a {@link ConformanceViolation} can blame. */
export type ConformanceCheckId = 'session-shape' | 'verified-claim' | 'assurance' | 'actor' | 'channel' | 'capabilities' | 'no-session' | 'revoke-session' | 'revoke-user-sessions' | 'server-session-create' | 'server-session-list' | 'api-key-verify';
/** One failed conformance assertion: which check, and what was wrong. */
export type ConformanceViolation = {
    readonly check: ConformanceCheckId;
    readonly message: string;
};
/**
 * Thrown by {@link assertProviderConformance} when a port fails any
 * check. Carries every violation found in the run, not just the first.
 */
export declare class ProviderConformanceError extends Error {
    readonly violations: readonly ConformanceViolation[];
    constructor(violations: readonly ConformanceViolation[]);
}
/** What one live-session scenario must narrow to. */
export type ExpectedConformanceSession = {
    readonly providerUserId: string;
    /** Expected `identity.email` claim value (and deprecated alias). */
    readonly email: string;
    /** Expected claim verification (and deprecated alias). */
    readonly emailVerified: boolean;
    readonly channel: SessionChannel;
    /** Expected `assurance.methods`; omit to accept any well-formed value. */
    readonly methods?: (readonly AuthMethod[] | 'unknown') | undefined;
    /** Expected `assurance.mfa`; omit to accept any well-formed value. */
    readonly mfa?: (boolean | 'unknown') | undefined;
    /** Expected actor; defaults to `null` (no impersonation). */
    readonly actor?: SessionActor | null | undefined;
    /** Expected `organizationId`; omit to accept any string-or-null. */
    readonly organizationId?: string | null | undefined;
    /** Expected `identity.anonymous`; defaults to `false`. */
    readonly anonymous?: boolean | undefined;
    /**
     * Expected `identity.phone` claim; omit to require the claim to be
     * well-formed only. Required as evidence when the port declares
     * the `'phone'` capability.
     */
    readonly phone?: {
        readonly value: string;
        readonly verified: boolean;
    } | null | undefined;
};
/** Scenario drivers for one freshly built port. */
export type ProviderConformanceScenarios = {
    /** A live session with a verified email claim. */
    readonly session: {
        readonly headers: Headers;
        readonly expected: ExpectedConformanceSession;
    };
    /**
     * A live session whose email claim is unverified: the claim and the
     * deprecated alias must both report `verified: false` (SEC-2).
     */
    readonly unverifiedSession?: {
        readonly headers: Headers;
        readonly expected: ExpectedConformanceSession;
    } | undefined;
    /**
     * A live impersonated session. Required when the port declares
     * `'impersonation'`, and supplying one requires that declaration:
     * declared capabilities must match behavior in both directions.
     */
    readonly impersonatedSession?: {
        readonly headers: Headers;
        readonly expectedActorProviderUserId: string;
    } | undefined;
    /** Headers carrying no session: `getSession` must return ok `null`. */
    readonly noSession?: {
        readonly headers: Headers;
    } | undefined;
    /**
     * Revoke observability. `'invalidates-session'` (the default)
     * requires a revoked session to read back as `null`;
     * `'best-effort'` is the honest declaration for stateless JWT
     * transports whose revocation cannot invalidate the token, and only
     * requires the revoke calls to succeed.
     */
    readonly revoke?: {
        readonly mode: 'invalidates-session' | 'best-effort';
    } | undefined;
};
/** One fresh port plus its scenario drivers. */
export type ProviderConformanceSubject = {
    readonly port: IdentityProviderPort;
    readonly scenarios: ProviderConformanceScenarios;
};
/**
 * Builds a fresh {@link ProviderConformanceSubject}. Called once per
 * check group (reads, `revokeSession`, `revokeUserSessions`), so each
 * mutating check starts from a live session.
 */
export type ProviderConformanceFactory = () => ProviderConformanceSubject | Promise<ProviderConformanceSubject>;
/**
 * Run every conformance check against a provider port factory and
 * return the violations (empty when the port conforms). Never throws
 * for a non-conforming port; a factory that itself throws propagates.
 */
export declare function checkProviderConformance(factory: ProviderConformanceFactory): Promise<readonly ConformanceViolation[]>;
/**
 * Assert that a provider port conforms: resolves when it does and
 * throws {@link ProviderConformanceError} listing every violation
 * otherwise. The one call each adapter's test suite makes.
 */
export declare function assertProviderConformance(factory: ProviderConformanceFactory): Promise<void>;

// ===== use-cases/apply-identity-event.d.ts =====
import { type ResultAsync } from 'neverthrow';
import type { RepositoryError, UserExistsError } from '../errors.js';
import type { IdentityEvent } from '../events.js';
import type { LoggerPort, UserRepository } from '../ports.js';
/**
 * Apply one provider {@link IdentityEvent} to the Payload user store
 * (CAP-14). Payload stays the single user store (invariant 1, D9):
 *
 * - `email-changed` syncs via `setEmail` only when the provider
 *   verified the new address; an unverified change is ignored.
 * - `email-verified` syncs the address when it differs.
 * - `identity-unlinked` and `user-deleted` remove only the identity
 *   row; the Payload user always remains.
 * - An event for a provider user with no linked Payload user is a
 *   no-op: that user was never linked (or was already unlinked).
 *
 * Never creates or deletes a Payload user.
 */
export declare function applyIdentityEvent(users: UserRepository, event: IdentityEvent, logger: LoggerPort): ResultAsync<void, RepositoryError | UserExistsError>;

// ===== use-cases/ensure-linked-user.d.ts =====
import { type ResultAsync } from 'neverthrow';
import { RepositoryError, UnverifiedEmailError, UnverifiedIdentityError, UserExistsError } from '../errors.js';
import { type LinkingPolicy } from '../linking.js';
import type { LoggerPort, ProviderId, ProviderProfile, UserRecord, UserRepository } from '../ports.js';
import type { Role } from '../roles.js';
export type EnsureLinkedUserDeps = {
    readonly users: UserRepository;
    readonly logger: LoggerPort;
};
export type EnsureLinkedUserInput = {
    readonly providerId: ProviderId;
    readonly profile: ProviderProfile;
    /** Roles for a newly created user. Existing users keep their roles. */
    readonly defaultRoles: readonly Role[];
    /**
     * Which verified claims may link (PORT-3). Defaults to
     * {@link defaultLinkingPolicy} (`verified-email`), preserving the
     * pre-PORT-3 behavior for callers that omit it.
     */
    readonly linkingPolicy?: LinkingPolicy | undefined;
};
/**
 * Bridge a provider identity to a Payload user, converging link and email.
 * Idempotent: an already-linked identity returns its user, syncing the
 * email when the provider reports a new verified address (an unverified
 * change is logged and kept stale, never applied); a matching email
 * links only when the provider verified it, otherwise the link is refused
 * with {@link UnverifiedEmailError} (no unverified-email account
 * takeover); an unknown email creates a fresh user with `defaultRoles`.
 *
 * The linking authority is the declared {@link LinkingPolicy}
 * (PORT-3) over the `identity` claims (PORT-1): the deprecated
 * `email`/`emailVerified` aliases on the profile are never read here.
 * Under the default policy a missing email claim refuses with
 * {@link UnverifiedIdentityError} and an unverified claim refuses
 * exactly as the flat flag did before — byte-identical to the
 * pre-claim gate (V1), which the existing tests pin as the regression
 * spec. The opt-in `verified-email-or-phone` policy adds exactly one
 * branch: a phone-only identity (no email claim) carrying a verified
 * phone claim links to an existing user by phone and never
 * provisions. An anonymous identity never links under any policy
 * (D3), not even to an already-linked user.
 *
 * `UserExistsError` escapes only on a lost create race: two concurrent
 * first touches for the same email. Callers retry the read once, which
 * then finds the winner's link.
 *
 * Verification gates creation exactly like linking: an unverified
 * provider email for an unknown address refuses with
 * {@link UnverifiedEmailError} instead of provisioning a full user,
 * so OAuth (or any flow yielding unverified sessions) cannot mint
 * default-role accounts.
 */
export declare function ensureLinkedUser(deps: EnsureLinkedUserDeps, input: EnsureLinkedUserInput): ResultAsync<UserRecord, RepositoryError | UnverifiedEmailError | UnverifiedIdentityError | UserExistsError>;

// ===== use-cases/register-user.d.ts =====
import { type ResultAsync } from 'neverthrow';
import { type Email } from '../email.js';
import { InvalidEmailError, RepositoryError, UserExistsError } from '../errors.js';
import type { LoggerPort, ProviderIdentityRef, UserRecord, UserRepository } from '../ports.js';
import type { Role } from '../roles.js';
export type RegisterUserDeps = {
    readonly users: UserRepository;
    readonly logger: LoggerPort;
};
export type RegisterUserInput = {
    readonly email: Email;
    readonly roles: readonly Role[];
    readonly identity?: ProviderIdentityRef | undefined;
};
/**
 * Register a new user. Pure domain flow over injected ports: rejects
 * duplicates with {@link UserExistsError}, persists via the repository,
 * and logs the outcome. Linking extra identities is a separate operation.
 */
export declare function registerUser(deps: RegisterUserDeps, input: RegisterUserInput): ResultAsync<UserRecord, RepositoryError | UserExistsError>;
/**
 * Registration-relevant subset of {@link UserRepository}: the two
 * operations registration needs. Plugin flows pass their repository
 * (which satisfies this structurally); provider flows implement it over
 * their own collections, whose row shapes (`auId`, `passwordHash`, …)
 * are not `NewUser`/`UserRecord` and must stay provider-owned.
 */
export type RegistrationStore<TCandidate, TCreated> = {
    findByEmail(email: Email): ResultAsync<TCreated | null, RepositoryError>;
    create(candidate: TCandidate): ResultAsync<TCreated, RepositoryError | UserExistsError>;
};
export type RegisterWithStoreDeps<TCandidate, TCreated> = {
    readonly store: RegistrationStore<TCandidate, TCreated>;
    readonly logger: LoggerPort;
};
export type RegisterWithStoreInput<TCandidate> = {
    /** Raw email; parsed (and normalized) here so stores see `Email` only. */
    readonly email: string;
    /** Build the store candidate from the parsed email. Runs only when no existing row is found, so expensive construction (password hashing) is skipped for duplicates. */
    readonly candidate: (email: Email) => TCandidate;
};
export type RegisteredWithStore<TCreated> = {
    readonly created: TCreated;
    readonly email: Email;
};
/**
 * Register through an injected store: the provider-registration flow.
 * Parses the email, rejects duplicates with {@link UserExistsError} (a
 * lost create race surfaces the same error — the caller learns the email
 * is taken, no retry needed), persists via the store, and logs the
 * outcome. Never throws; storage failures arrive as
 * {@link RepositoryError} for the caller to map to its own taxonomy.
 */
export declare function registerWithStore<TCandidate, TCreated>(deps: RegisterWithStoreDeps<TCandidate, TCreated>, input: RegisterWithStoreInput<TCandidate>): ResultAsync<RegisteredWithStore<TCreated>, InvalidEmailError | UserExistsError | RepositoryError>;

// ===== version.d.ts =====
/**
 * One supported upstream version range. A band is a list of these:
 * version lines we have actually validated (bisected floor, tested
 * ceiling), never a hopeful caret. Prerelease versions match only
 * via an `exact` entry — the B10 canary leak came from semver
 * prerelease ordering silently admitting untested builds.
 */
export type VersionRange = {
    readonly min: string;
    readonly maxExclusive: string;
} | {
    readonly exact: string;
};
/** Parsed semver core; `prerelease` is `null` for plain releases. */
export type ParsedVersion = {
    readonly major: number;
    readonly minor: number;
    readonly patch: number;
    readonly prerelease: string | null;
};
/** Parse `major.minor.patch[-prerelease][+build]`; `null` if unparsable. */
export declare function parseVersion(version: string): ParsedVersion | null;
/** True when `version` falls inside any range of the band. */
export declare function isSupportedVersion(version: string, band: readonly VersionRange[]): boolean;
/** Human-readable band, for error messages (`>=1.7.6 <1.8.0 || 4.0.0-beta.0`). */
export declare function describeBand(band: readonly VersionRange[]): string;
/**
 * Startup version assertion (upgrade resilience Phase 1): an
 * upstream version outside the validated band fails at wiring time
 * with {@link UnsupportedVersionError} — before any collection
 * write — instead of surfacing later as a raw SQL error, a missing
 * adapter method, or a silently wrong schema.
 */
export declare function assertSupportedVersion(packageName: string, installedVersion: string, band: readonly VersionRange[]): void;
/**
 * Read the installed version of an upstream package from its
 * `package.json` manifest. Resolution follows the caller: pass the
 * importing module's `import.meta.url` as `fromUrl` so the manifest
 * resolves through the caller's own dependency tree (under strict
 * linkers, core cannot see the caller's deps). The manifest is
 * located by resolving the package entry and walking up — some
 * packages (`@auth/core`) block the `./package.json` subpath in
 * their `exports`. Returns `null` when nothing resolves (exotic
 * bundling) — callers decide whether that skips the assertion (with
 * a warning) or fails.
 */
export declare function readInstalledVersion(packageName: string, fromUrl?: string): string | null;
