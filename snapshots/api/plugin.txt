import { ChannelNotAdmittedError, csrfHeaderName, RepositoryError, SessionExpiredError, UserSuspendedError, type LoggerPort, type SessionChannel, type ResolvedSessionActor, type SessionInvalidError } from '@payload-auth-link/core';
import type { CollectionConfig, PayloadRequest } from 'payload';
export { csrfHeaderName };
/**
 * Loud logging for session verification failures. Storage outages are
 * errors with their cause; dead tokens are warnings. Call before
 * returning `{ user: null }` / 401 so every denial is traceable.
 */
export declare function logVerifyFailure(logger: LoggerPort, scope: string, error: SessionInvalidError | SessionExpiredError | ChannelNotAdmittedError | UserSuspendedError | RepositoryError): void;
/** Strict same-origin comparison against `baseUrl`. Never throws. */
export declare function isSameOrigin(value: string, baseUrl: string): boolean;
/**
 * CSRF proof for state-changing endpoints: a matching `Origin` header or the
 * custom header. Requests with neither (plain cross-site forms, blind curl)
 * fail closed.
 */
export declare function passesCsrf(headers: Headers, baseUrl: string): boolean;
/**
 * Channel-aware CSRF proof (CAP-13b). Cookie sessions are ambient
 * credentials — a cross-site page can ride them — so they always
 * need {@link passesCsrf}. A presented `Authorization: Bearer`
 * token is NOT ambient (no browser attaches it cross-site), so when
 * the deployment explicitly admits the `bearer` channel, the origin
 * check is skipped for bearer-presented requests ONLY. Anything
 * else (no header, bearer not admitted) keeps the cookie rules.
 * Note this only decides CSRF: whether the bearer session itself is
 * admitted is the session service's channel gate, which still runs.
 */
export declare function passesCsrfForChannel(headers: Headers, baseUrl: string, admitChannels: readonly SessionChannel[]): boolean;
/**
 * JSON envelope for endpoint responses. Session state must never sit in
 * any cache (browser, proxy, or framework): every response carries
 * `no-store`, including errors — a cached 401 would pin a signed-out
 * user, a cached 200 a signed-in one.
 */
export declare function jsonResponse(data: unknown, status: number): Response;
/** 401 envelope; the detail never leaks which check failed. */
export declare function unauthorized(detail?: string): Response;
/** 403 envelope for authenticated-but-denied requests. */
export declare function forbidden(detail?: string): Response;
/**
 * Server-side impersonation state of a Payload request. This is the
 * AUTHORIZATION view, deliberately separate from the client wire
 * parser (`parseSessionUser`): it reads only the `actor` field the
 * auth strategy projects onto `req.user`, requires no wire-valid
 * identity fields around it, and is FAIL-CLOSED — a present,
 * non-null `actor` that cannot be parsed is `malformed-actor`,
 * which every deny/audit path treats as impersonating. A parse
 * failure can never launder an impersonated request into a plain
 * one.
 */
export type ImpersonationState = {
    readonly kind: 'none';
} | {
    readonly kind: 'actor';
    readonly actor: ResolvedSessionActor;
} | {
    readonly kind: 'malformed-actor';
};
/** Fail-closed server extraction; see {@link ImpersonationState}. */
export declare function impersonationStateFromRequest(req: PayloadRequest): ImpersonationState;
/**
 * Whether this request is impersonated, fail-closed: a well-formed
 * actor OR a present-but-malformed one. Authorization denies must
 * use this (or {@link impersonationStateFromRequest}), never a
 * null-check on {@link impersonationActorFromRequest}.
 */
export declare function isImpersonating(req: PayloadRequest): boolean;
/**
 * The impersonating operator behind a Payload request, or `null`
 * when there is none — or none that can be honestly named. For
 * ATTRIBUTION (audit fields, display) only: a malformed actor
 * yields `null` here yet still denies via {@link isImpersonating},
 * so never gate access on this function's nullness.
 */
export declare function impersonationActorFromRequest(req: PayloadRequest): ResolvedSessionActor | null;
/** Collection operations {@link denyWhileImpersonating} can guard. */
export type ImpersonationOperation = 'admin' | 'create' | 'delete' | 'read' | 'readVersions' | 'unlock' | 'update';
/**
 * Default operations denied while impersonating (CAP-8b / D5):
 * admin-panel access and every write — which includes role changes,
 * since roles change through user updates — while reads stay allowed
 * so an operator can see what the user sees. The plugin applies the
 * admin + roles-field half of this default to the users collection
 * itself; `denyWhileImpersonating` extends it to your collections.
 */
export declare const defaultDeniedWhileImpersonating: readonly ImpersonationOperation[];
/** The collection `access` map {@link denyWhileImpersonating} returns. */
export type DenyWhileImpersonatingAccess = NonNullable<CollectionConfig['access']>;
/**
 * Build a collection `access` map that denies the given operations
 * while an operator is impersonating the request's user (CAP-8b,
 * decision D5: configurable, default read-mostly). Non-denied
 * operations — and every operation for a non-impersonated request —
 * return `true`, so compose with your own rules instead of replacing
 * them: wrap the slot you need (`access.update` below) or call
 * {@link impersonationActorFromRequest} inside your access function.
 * This helper is a deny layer, never a grant.
 *
 * ```ts
 * const impersonation = denyWhileImpersonating()
 * access: {
 *   admin: impersonation.admin,
 *   update: ({ req }) =>
 *     !isImpersonating(req) && myOwnUpdateRule(req),
 * }
 * ```
 *
 * Payload evaluates access per operation slot, so there is no
 * single-function form: the map is the honest shape. Unknown
 * operation names (from JS callers) fail fast with
 * `InvalidConfigError` instead of silently guarding nothing.
 */
export declare function denyWhileImpersonating(operations?: readonly ImpersonationOperation[]): DenyWhileImpersonatingAccess;
/**
 * Server-side MFA state of a Payload request (CAP-6), read from the
 * `assurance` fact the auth strategy projects onto `req.user`.
 * FAIL-CLOSED like the impersonation extraction: absent, `'unknown'`,
 * or malformed assurance is "not satisfied" — only the literal
 * `mfa: true` passes. Never route this through the client wire
 * parser; this reads the server-projected user only.
 */
export declare function sessionMfaSatisfied(req: PayloadRequest): boolean;
/**
 * Build a collection `access` map that denies the given operations
 * unless the request's session carries MFA proof (CAP-6). The
 * companion to {@link denyWhileImpersonating} and composed the same
 * way — a deny layer, never a grant. The default denies the admin
 * panel and every write; the plugin's `policies.requireMfaForRoles`
 * covers role-gated surfaces by withholding the roles themselves,
 * while this helper guards collections whose sensitivity is not
 * role-shaped. Unknown operation names fail fast.
 */
export declare function denyWithoutMfa(operations?: readonly ImpersonationOperation[]): DenyWhileImpersonatingAccess;

     * (`admin.authBase`).

    readonly authBase: string;

export type MappingContext = {
    readonly map: CollectionMap;
    readonly defaultRole: Role;
    readonly logger: LoggerPort;
    readonly clock: ClockPort;
    /**
     * Governance/identity field names (CAP-7/CAP-9). Optional so direct
     * mapping callers keep working; absent means the plugin defaults
     * (`phone`, `suspendedUntil`, `suspendReason`).
     */
    readonly userFields?: {
        readonly phoneField: string;
        readonly suspendedUntilField: string;
        readonly suspendReasonField: string;
    } | undefined;
    /**
     * Stored value → canonical role (issue #56). Present when the site
     * maps role values; reads translate through it before validation, so
     * a payload-auth layout keeps storing e.g. `frontPlugin` while the
     * plugin reasons about `front-plugin`.
     */
    readonly roleValuesReverse?: Readonly<Record<string, string>> | undefined;

     * Canonical role → stored value (issue #56); present for writes.
     * Unmapped roles store their canonical form, so this is total when

     * may omit it to keep the format-only narrowing.

    readonly configuredRoles?: ReadonlySet<Role> | undefined;

     */
    readonly roleValues?: Readonly<Record<string, string>> | undefined;
    /**
     * Admin panel surface (issue #55): `'provider'` swaps Payload's
     * local login view for the provider-backed one — email+password
     * through the provider's sign-in endpoint plus the break-glass
     * local fallback form (unlinked operator accounts only). The view
     * is registered as a Payload custom view; the import map must
     * include `@payload-auth-link/plugin/admin#ProviderLoginView` (the
     * doctor checks for it).
     */
    readonly admin?: {
        readonly loginView?: 'provider' | undefined;
        /**
         * The app's sign-in page the provider login view links
         * social-only accounts to. Defaults to `/sign-in`.
         */
        readonly signInPath?: string | undefined;
        /**
         * App-relative mount of the cookie-transport provider's
         * routes (Better Auth's `basePath` as served by the app) —
         * the provider form posts to `${authBase}/sign-in/email`.
         * Defaults to `/api/auth`.
         */
        readonly authBase?: string | undefined;
    } | undefined;
    /**
     * Coexistence window (issue #51/#72): the same window you pass to
     * the provider ENGINE — `createBetterAuthEngine({ legacySessions })`
     * — which is the option that actually honors legacy payload-auth
     * session cookies (read-only) until the ISO `until` date. The
     * plugin cannot enable coexistence itself; this option exists to
     * ASSERT the engine's window at config time: wiring throws when
     * no engine declares one, when the two disagree, or when
     * `cookieNames` collides with the engine's current session cookie
     * names. Setting it without the engine input is a config error,
     * not a silent no-op.
     */
    readonly legacySessions?: {
        readonly until: string;
        readonly cookieNames: readonly string[];
    } | undefined;
    /**
     * Server session creation gate (issue #54): controls
     * `authLinkServer(payload).createSessionFor`. `'dev-only'` (the
     * default) enables it ONLY on a development runtime (`NODE_ENV ===
     * 'development'` — production, test, staging, and unset all
     * refuse); `true` enables it everywhere; `false` disables it
     * everywhere. Reads, listings, revocations, and API-key
     * verification are never gated — only minting is.
     */
    readonly allowServerSessions?: 'dev-only' | true | false | undefined;
    /**
     * What `req.user` carries (issue #57): `'projection'` (the default)
     * keeps the stable session projection `{ id, collection, email,
     * roles, … }`; `'document'` loads the FULL user document once per
     * request (the `req.context` memo) so access functions reading other
     * fields keep working unchanged. The security-relevant projected
     * fields stay authoritative either way: `roles` after the MFA
     * projection, `actor`/impersonation markers, suspension.
     */
    readonly reqUser?: 'projection' | 'document' | undefined;
    /**
     * Which form `req.user.roles` carries (issue #56): `'canonical'`

import { type Role } from '@payload-auth-link/core';

/**
 * Payload-owned organizations (CAP-11, decision D1): `organizations`
 * and `memberships` collections (one role per user per org) that work
 * with every provider, because membership is a property of Payload
 * users — and Payload owns the users. A provider-native org store
 * (e.g. BA's `organization` plugin) stays a provider feature and is
 * never Payload authorization.
 *
 * The plugin stores no session state for the "active" org (invariant
 * 1): the host carries it (request param, cookie, URL) and
 * authorization asks {@link requireOrgRole} per request.
 *
 * Role semantics: `orgRoles` is ordered by privilege. The creator of
 * an organization receives the FIRST role (the owner role —
 * `'owner'` by default). Management operations (invite, role
 * change, org update/delete) require the first role, or `'admin'`
 * when present in the list.
 *
 * Membership governance (all enforced server-side, every channel):
 * - a membership's `organization` and `user` are immutable once
 *   created — a row can never be relocated into another tenant or
 *   re-pointed at another user (field access AND a `beforeChange`
 *   guard, so trusted Local-API writes are held to it too; the only
 *   sanctioned transition is the plugin's own invite acceptance,
 *   which binds the accepting user to a pending invitation);
 * - role writes (create, update, invite) require a manager of the
 *   row's organization and never exceed the caller's own role —
 *   `orgRoles` order is the rank, so an `admin` can neither promote
 *   anyone (themselves included) to owner nor touch an owner's row;
 * - an organization always keeps at least one active owner: demoting
 *   or removing the last one is refused with a typed 409;
 * - deleting an organization cascades its memberships and pending
 *   invitations in the same transaction;
 * - every membership/organization write, invite, and acceptance is
 *   refused while an operator impersonates the caller, and the
 *   attempt is audited (CAP-8b).
 */

    readonly id: string;
    readonly organizationId: string;
    readonly userId: string | null;

    readonly status: string;
