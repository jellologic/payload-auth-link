// ===== codemods.d.ts =====
/**
 * Codemods (issues #53, #70): rule-based rewrites for the call sites a
 * payload-auth app must migrate, matched on the TypeScript AST and
 * applied as text splices over the original ranges (no printer —
 * formatting survives).
 *
 * Each rule rewrites only what it can FULLY rewrite and reports the
 * rest as manual sites ({@link findManualSites}); a half-rewritten
 * call site is worse than an explicit one. In particular, a file with
 * ANY unrewritable `getPayloadAuth` reference keeps its whole
 * `get-payload-auth` state untouched, so the output never references a
 * symbol no import provides.
 */
export type CodemodRule = {
    readonly id: string;
    readonly description: string;
    /** Apply to one file's content; returns the new content (same string when untouched). */
    readonly apply: (content: string) => string;
};
export type CodemodFileResult = {
    readonly file: string;
    readonly changed: boolean;
    /** Rules that fired on this file. */
    readonly applied: readonly string[];
    /** Matches the rules could not fully rewrite (reported, never touched). */
    readonly manual: readonly string[];
    /** The rewritten content — equal to the input when nothing changed. */
    readonly after: string;
};
export declare const codemodRules: readonly CodemodRule[];
/**
 * Report call sites the rules cannot fully rewrite (issue #70.4):
 * EVERY remaining `getPayloadAuth` reference (calls, type references,
 * the import itself), every surviving `payload.betterAuth` accessor
 * (including `.betterAuth.api.<method>` and `$context`), and every
 * `internalAdapter`. Nothing is silently ignored — the doctor counts
 * these references, so the codemod must account for them.
 */
export declare function findManualSites(content: string): readonly string[];
/** Apply every rule to one file; reports what fired and what needs a human. */
export declare function applyCodemods(file: string, content: string): CodemodFileResult;

// ===== doctor/run-doctor.d.ts =====
import type { BehaviourChange, DatabaseFinding, DoctorIssue, DoctorReport } from './types.js';
/**
 * The read-only doctor (issue #52): point it at a payload-auth app and
 * get the exact list of what breaks before touching anything. It loads
 * the app's config module, inspects the users collection and Better
 * Auth plugins, scans the source tree for call sites that must
 * migrate, and (with a database URL) reports row counts, unverified
 * users, and orphans. It never writes.
 *
 * Never crashes (issue #71): a config that cannot even be IMPORTED
 * (the mid-upgrade state — better-auth 1.7 installed while the config
 * still calls `oidcProvider`) is a `config-unloadable` blocker, not a
 * crash. The static scans (config text, source tree, database) still
 * run and still catch what the load cannot.
 */

    readonly configPath: string;

    readonly srcDir?: string | undefined;
    /** Postgres connection string. Defaults to `DATABASE_URL`. */
    readonly databaseUrl?: string | undefined;
    /**
     * The `--source-map` JSON (issue #68): model → source table name, or
     * `null` for "model not installed". Shapes the database section, on
     * top of the layout derived from the loaded payload-auth config
     * (issue #113) — explicit entries win.
     */
    readonly sourceMapJson?: string | undefined;
    /** Config loader, injectable for tests. Defaults to dynamic import. */
    readonly loadModule?: ((path: string) => Promise<unknown>) | undefined;
};

 * app feels on day one. Static by nature; kept next to the doctor so
 * the report is one artifact.

// ===== importer/apply.d.ts =====
import type { Payload } from 'payload';
import { type IdentityCheck } from './guards.js';
import { type SourceUser } from './source.js';
import { type ImportManifest, type ImportResult, type PlanRow, type Unmappable } from './types.js';
/**
 * Apply side (issue #50): copy payload-auth rows into the
 * payload-auth-link collections through the Payload Local API.
 *
 * - Password hashes, OAuth tokens, API-key hashes, and jwks carry over
 *   VERBATIM (payload-auth used Better Auth's own hashing).
 * - Row timestamps (`createdAt`/`updatedAt`) carry from the source rows
 *   for every model whose source table has them (issue #88).
 * - Better Auth plugin user state carries (issue #88): admin-plugin
 *   `banned`/`banReason`/`banExpires`/`role` and the anonymous plugin's
 *   `isAnonymous`, whenever the source has the column AND the target
 *   `ba-users` collection declares the field. A banned or anonymous
 *   source user whose state the target cannot hold is UNMAPPABLE — the
 *   run aborts before writing; a banned user never imports unbanned.
 * - `linkable` + `linkableEmail` are granted only with
 *   `grantLinkableFromVerified`, and then from VERIFIED emails only
 *   (grant-only: an existing proof is never revoked). Every imported
 *   user resolves through their pre-linked identity without it.
 * - Sessions default to `carry` so nobody is logged out: the carried
 *   rows keep their tokens, and with the same `advanced.cookiePrefix`
 *   the existing cookies keep working.
 * - Idempotent and resumable: every target row carries a deterministic
 *   `baId` derived from the source id (`acct-<source id>`), so a
 *   re-run after a crash finds its own earlier rows and skips them.
 * - Every created row and every host field write lands in the
 *   manifest — the FIRST captured prior value per field wins, sealed —
 *   so `rollback` is exact and the source tables stay untouched.
 * - Every host field write is marked with the kit's migration context
 *   (`MIGRATION_CONTEXT`, see host-writes.ts) so host hooks can skip
 *   their outward side effects for migration writes (issue #123).
 */
export type ApplyOptions = {
    readonly payload: Payload;
    readonly sourceConnectionString: string;
    readonly apply: boolean;
    readonly sessions: 'carry' | 'drop';
    readonly manifest: ImportManifest | null;
    /** Written (merged) on `apply`; required when `apply` is true. */
    readonly manifestPath: string;
    readonly sourceCounts: Readonly<Record<string, number>>;
    readonly sourceDatabase: string;
    /** Target collection slugs; defaults must match the engine wiring. */
    readonly slugs?: Readonly<Record<string, string>> | undefined;
    /**
     * Source-side model → table map (issue #68). Defaults to
     * payload-auth's layout; a `null` model is treated as not installed.
     */
    readonly sourceTables?: Readonly<Record<string, string | null>> | undefined;
    /** Injectable clock for deterministic `linkedAt` stamps. */
    readonly now?: () => Date;
    readonly log?: (message: string) => void;
    /**
     * Accept an unsigned or 0.6.0 manifest (`--allow-unsigned-manifest`) —
     * for the dry run only: such a manifest is never resumed by `--apply`
     * (GHSA-crrr-3cg7-f6jw); roll it back or finalize it.
     */
    readonly allowUnsignedManifest?: boolean | undefined;
    /**
     * Resume a fully signed manifest written by migrate 0.7.0 or earlier,
     * which names no database (`--allow-unbound-manifest`): the operator
     * vouches it was written against THIS database. The resume binds it.
     */
    readonly allowUnboundManifest?: boolean | undefined;
    /**
     * Grant the linking proof (`linkable` + `linkableEmail`) from the
     * source `emailVerified` flag (`--grant-linkable-from-verified`).
     * Off by default: that flag can be stale for an address changed
     * after it was verified, and the proof lets ANOTHER provider's
     * sign-in for the address link to the document (GHSA-v6mr).
     */
    readonly grantLinkableFromVerified?: boolean | undefined;
    /**
     * How this environment identifies itself (`--environment-id`,
     * `--database-moved`, `--new-database-identity`;
     * GHSA-xw9m-ph95-84rx).
     */
    readonly identity?: IdentityCheck | undefined;
};
/**
 * Plugin user state the target must be able to hold (issue #88),
 * checked BEFORE any write: a banned user whose target has no `banned`
 * field would import unbanned; an anonymous user without `isAnonymous`
 * would import as a full account.
 */
export declare function pluginStateUnmappable(users: readonly SourceUser[], targetFields: ReadonlySet<string>, targetSlug: string): Unmappable[];
export declare function applyImport(options: ApplyOptions): Promise<ImportResult>;
/** Collapse per-row plan entries into one line per model (dry-run print). */
export declare function summarizePlanRows(rows: readonly PlanRow[]): PlanRow[];

// ===== importer/binding.d.ts =====
import type { Payload } from 'payload';
import { type IdentityCheck, type RunRecord } from './guards.js';
import type { ImportManifest, ManifestKind, TargetSlugMap } from './types.js';
/**
 * Manifest ↔ database binding (GHSA-crrr-3cg7-f6jw).
 *
 * A manifest is a file; the database it drives is not. Every write the
 * kit makes from a manifest therefore checks the DATABASE's own record
 * first, kept in Payload's key-value store:
 *
 * - a database id, minted on the first `--apply` (or `adopt-prelink
 *   --apply`), which every version-4 manifest names inside its
 *   signature — a manifest from another environment (a rehearsal clone
 *   sharing `PAYLOAD_SECRET`) names another id and is refused;
 * - one record per run (a random nonce, also signed into the manifest):
 *   open, rolled back, or finalized. Only an OPEN run is resumed or
 *   rolled back, so a stale copy of a rolled-back run's manifest
 *   changes nothing. At most ONE run is open at a time (issue #116):
 *   two open runs would each hold rows the other's rollback or finalize
 *   cannot see.
 *
 * The database id alone does not tell a database from its clone (a
 * copy taken after the first `--apply` carries the same record), so the
 * record also keeps a server fingerprint (Postgres) and the operator's
 * environment id — see `databaseIdentity` (GHSA-xw9m-ph95-84rx).
 *
 * Unsigned manifests (pre-0.6.0) and 0.6.0 manifests whose signature
 * does not cover `finalized` are TERMINAL: they may be rolled back or
 * finalized, never resumed — a resume would re-sign their (unverifiable)
 * entries into a manifest every later command trusts.
 */
/** The database and run a manifest write belongs to. */

    readonly databaseId: string;
    readonly runId: string;
    readonly createdAt: string;
};

}, action: string, identity?: IdentityCheck): Promise<RunRecord>;

/**

 * resume:
 *
 * - another command's manifest, an edited one, a finalized one;
 * - an unsigned or 0.6.0 manifest (terminal: roll it back or finalize);
 * - one holding plaintext prior values (only a pre-0.6.0 manifest
 *   writes them; a signed manifest carrying any was laundered by an
 *   earlier resume);
 * - created rows outside the kit's own targets;
 * - a version-4 manifest from another database, or whose run this
 *   database does not hold open. A run the database records as rolled
 *   back starts FRESH: the manifest's entries are not carried;
 * - a 0.7.0 (unbound) manifest without `allowUnboundManifest`, or on a
 *   database that records a finalized migration;
 * - a FRESH run while another run is open (issue #116);

/**
 * Host-document field writes with exact rollback (issues #50, #88).
 *
 * Capture: before the importer (or `adopt-prelink`) writes a host
 * field, its CURRENT value is captured into the manifest — once per
 * document field, the FIRST capture wins, so a second `--apply` can
 * never overwrite the pre-import value with imported state. The
 * document's `updatedAt` is captured the same way, because the Local
 * API write bumps it.
 *
 * Privacy: captured values (identity rows and `linkableEmail` carry
 * email addresses) are sealed with AES-256-GCM under a key derived
 * from the target app's Payload `secret`; the manifest stores no value
 * in the clear. Rollback needs the same secret — it boots the same
 * app. Each value is bound (GCM associated data) to its entry AND to
 * the database id and run nonce of the manifest that captured it
 * (GHSA-xw9m-ph95-84rx), so it opens in no other manifest.
 *
 * Restore: rollback writes the captured values back through the
 * DATABASE ADAPTER (`payload.db.updateOne`), not the Local API
 * operation: the operation would stamp a fresh `updatedAt` and re-run
 * hooks, so the "restored" row would differ from the original. The
 * adapter write sets exactly the captured fields — `updatedAt`
 * included — on every adapter (Postgres, SQLite, Mongo). What it
 * cannot preserve is documented in docs/migrating-from-payload-auth.md
 * (the forward Local API write re-creates hasMany child rows such as
 * `users_role` with new row ids; their values and order are intact).
 */
/**
 * The `req.context` key the kit marks every host write with (issue
 * #123). The kit's host writes run through the HOST app's Local API, so
 * the users collection's `beforeChange` / `afterChange` hooks fire for
 * each one — `from-payload-auth --apply`, `adopt-prelink --apply`, and
 * every rollback restore alike. A host hook that triggers outward side
 * effects (a CRM sync, an email, a webhook, re-indexing) skips them for
 * migration writes by checking this key:
 *
 * ```ts
 * import { MIGRATION_CONTEXT } from '@payload-auth-link/migrate'
 *
 * // in a users-collection hook:
 * if (req.context?.[MIGRATION_CONTEXT] !== undefined) return
 * ```
 */

     * Accept a manifest without a full signature (`--allow-unsigned-manifest`):
     * a pre-0.6.0 manifest without a MAC, or a 0.6.0 manifest whose MAC
     * does not cover `finalized`. The operator vouches that nobody edited
     * it and that it was written against THIS database.
     */
    readonly allowUnsignedManifest?: boolean | undefined;
    /**
     * Let a rollback driven by an UNSIGNED (pre-0.6.0) manifest delete the
     * rows it lists (`--allow-unsigned-deletes`). Without it, an unsigned
     * manifest may only restore sealed field values.
     */
    readonly allowUnsignedDeletes?: boolean | undefined;
    /**
     * Restore version-1 PLAINTEXT prior values (`--allow-plaintext-restore`,
     * GHSA-crrr-3cg7-f6jw). Nothing vouches for a value stored in the
     * clear — not even a signature an older kit added when it resumed the
     * manifest — so the operator reviews them and opts in separately.
     */
    readonly allowPlaintextRestore?: boolean | undefined;
    /**
     * Accept a fully signed manifest written by migrate 0.7.0 or earlier,
     * which names no database (`--allow-unbound-manifest`): the operator
     * vouches it was written against THIS database.
     */
    readonly allowUnboundManifest?: boolean | undefined;
    /** How this environment identifies itself (GHSA-xw9m-ph95-84rx). */
    readonly identity?: IdentityCheck | undefined;
};
/**
 * Rollback (issue #50): delete the created rows in reverse order and
 * restore every captured field value. The manifest records exactly
 * what to undo — nothing else is touched, and the source tables were
 * never modified.
 *
 * Before anything is written, rollback refuses: a finalized manifest,
 * or a database that records ANY finalized migration (the marker
 * `finalize` writes — whatever kind the manifest claims; GHSA-crrr);
 * a key-value store it cannot read (fail closed); a manifest that
 * fails its signature; an UNSIGNED manifest on a database that holds a
 * migration id (GHSA-xw9m-ph95-84rx); a database whose migration
 * record was copied from another one; a captured value sealed for
 * another run; a version-4 manifest naming another database,
 * or a run this database does not hold open; a 0.7.0 manifest (no
 * database binding) without `allowUnboundManifest`; created rows
 * outside the import's own target collections (signed or not), or
 * whose row no longer carries the importer's deterministic `baId`;
 * any delete driven by an unsigned manifest without
 * `allowUnsignedDeletes`; and any plaintext prior value without
 * `allowPlaintextRestore`. A row already absent is fine (the goal is
 * "absent"); a delete that FAILS (a foreign key, a hook, a permission)
 * aborts the rollback with the rows named, before any field is
 * restored or the manifest reset — so it can be re-run once fixed.
 * Deletes and restores share one database transaction where the
 * adapter offers one, so a failed rollback changes nothing.
 * Success records the run as rolled back in the database, so no copy
 * of this manifest drives a second rollback.
 */
export declare function rollbackImport(options: RollbackOptions): Promise<{
    deleted: number;
    restored: number;
}>;
/** Options of {@link finalizeManifest}. */
export type FinalizeOptions = {
    readonly payload: Payload;
    readonly manifest: ImportManifest;
    /** Where the finalized manifest is written. */
    readonly path: string;
    readonly now?: (() => Date) | undefined;
    /** Accept a manifest without a full signature (`--allow-unsigned-manifest`). */
    readonly allowUnsignedManifest?: boolean | undefined;
    /** How this environment identifies itself (GHSA-xw9m-ph95-84rx). */
    readonly identity?: IdentityCheck | undefined;
};
/**
 * Finalize (issue #50): the point of no return. Records a one-way
 * marker in the target database (Payload's key-value store) that every
 * later rollback checks independently of the file, then marks the
 * manifest `finalized`. A signed manifest is re-signed (the signature
 * covers the flag); a version-4 manifest must name THIS database and a
 * run it does not record as rolled back, and its run is recorded as
 * finalized. An unsigned or 0.6.0 manifest is never re-signed
 * (GHSA-crrr-3cg7-f6jw): its flag is set, and the database marker is
 * what every later command trusts. Idempotent: finalizing a finalized
 * manifest re-records the marker.
 *
 * Refused (GHSA-xw9m-ph95-84rx, issue #116): an UNSIGNED manifest on a
 * database that holds a migration id — a forged one would otherwise
 * write the marker and block every rollback for good — and any
 * finalize while ANOTHER open run may still hold rows.
 */
export declare function finalizeManifest(options: FinalizeOptions): Promise<ImportManifest>;

/**
 * The adopt-mode pre-link (issues #49, #88): `migrate adopt-prelink`.
 *
 * In adopt mode the users collection IS Better Auth's user table
 * (`schemaMap.user`, `idStrategy: 'payload'`), so Better Auth's user id
 * for a document is the document id. What adopt mode does NOT do by
 * itself is link the plugin side: without an identity row, the first
 * provider sign-in reaches JIT linking, which links an existing
 * document only on the verified-email proof (`linkable === true` with
 * `linkableEmail` equal to the document email).
 *
 * This command mirrors exactly what the importer writes on the host
 * users (issue #69):
 *
 * - ONE `better-auth` identity per user, `providerUserId` = the
 *   document id (other providers' identity rows are kept);
 * - with `grantLinkableFromVerified` (`--grant-linkable-from-verified`)
 *   only: `linkable: true` + `linkableEmail: <email>` for users whose
 *   verified-email field (`emailVerified` on payload-auth layouts) is
 *   `true` — grant-only, never revoked. Off by default: that flag is a
 *   mirror that can be stale for an address changed after it was
 *   verified, and the proof lets ANOTHER provider's sign-in for the
 *   address link to the document (the GHSA-v6mr outcome). The
 *   better-auth identity resolves every user without it.
 *

 * (first capture wins, sealed, mode 0600) so `rollback --manifest`
 * restores the documents exactly, `updatedAt` included. The write
 * itself keeps each document's `updatedAt` (issue #116); the Local
 * API's full-document write does re-create hasMany child rows (such as
 * `users_role`) with new surrogate row ids — values and order intact.
 * The write carries the kit's migration context marker
 * (`MIGRATION_CONTEXT`, host-writes.ts) so host hooks can skip their
 * outward side effects (issue #123).
 *
 * The users collection is the ONLY one this command writes: the
 * invitations table is not among its written tables, so invitation
 * rows carrying the legacy JSON-encoded role strings (issue #142) are
 * NOT normalized here — rewrite them by hand (the doctor reports them
 * as `legacy-invitation-roles`).
 */

    readonly payload: Payload;
    readonly apply: boolean;
    readonly manifest: ImportManifest | null;
    readonly manifestPath: string;

    /** The users' verified-email checkbox. Defaults to `emailVerified`. */
    readonly verifiedField?: string | undefined;
    /** Email field. Defaults to `email`. */
    readonly emailField?: string | undefined;
    readonly now?: () => Date;
    readonly log?: (message: string) => void;
    /**
     * Accept an unsigned or 0.6.0 manifest (`--allow-unsigned-manifest`) —
     * for the dry run only: such a manifest is never resumed by `--apply`.
     */
    readonly allowUnsignedManifest?: boolean | undefined;
    /** Resume a signed 0.7.0 manifest that names no database (`--allow-unbound-manifest`). */
    readonly allowUnboundManifest?: boolean | undefined;
    /**
     * Grant the linking proof from the verified-email field
     * (`--grant-linkable-from-verified`). Off by default — see above.
     */
    readonly grantLinkableFromVerified?: boolean | undefined;
    /** How this environment identifies itself (GHSA-xw9m-ph95-84rx). */
    readonly identity?: IdentityCheck | undefined;
};

    readonly users: number;

    readonly linked: number;

 * changed, so `rollback` is exact. The manifest is the run's contract —
 * `verify`, `rollback`, and `finalize` all read it.

 * command checks them against the database before it writes, so a
 * manifest from another environment or another run is refused.

    readonly sourceKey: string;
};

    readonly before?: string | null | undefined;

     * The prior value, JSON-serialized and sealed with the target app's
     * Payload secret (version 2, issue #88), restored by rollback. Only
     * the FIRST capture per document field is kept, so a rollback after
     * any number of `--apply` passes restores the pre-import state.
     */

};

     * Version 4: the id of the database this manifest belongs to (stored
     * in that database's Payload key-value store on the first `--apply`).
     * Absent on manifests written by migrate 0.7.0 and earlier.
     */
    readonly databaseId?: string | undefined;
    /**
     * Version 4: this run's random nonce. The database records each run's
     * state (open, rolled back, finalized); a manifest whose run the
     * database does not hold as open is never resumed or rolled back.
     */
    readonly runId?: string | undefined;
    /**
     * Version 3: HMAC over the whole manifest, keyed from the target
     * app's Payload secret. `v2.` MACs cover every field; 0.6.0's `v1.`
     * MACs left `finalized` out. See `manifest-mac.ts`.
     */
    readonly mac?: string | undefined;
};
/** Per-model plan row for the dry run. */
export type PlanRow = {
    readonly model: string;
    readonly sourceTable: string;
    readonly targetCollection: string;
    readonly rows: number;
    readonly skipped: number;
    readonly unmappable: readonly string[];
};
/** Why a row cannot be imported — the run aborts rather than skips. */
export type Unmappable = {
    readonly model: string;
    readonly key: string;
    readonly reason: string;
};
export type ImportPlan = {
    readonly rows: readonly PlanRow[];
    readonly unmappable: readonly Unmappable[];
    readonly sessions: 'carry' | 'drop';
    readonly warnings: readonly string[];
};
export type ImportResult = {
    readonly created: number;
    readonly skipped: number;
    readonly fieldWrites: number;
    readonly plan: ImportPlan;
};

// ===== kv-adapter.d.ts =====
/**
 * How the app's Payload key-value store is backed (issue #132). The
 * migration records its database identity, its runs, and the finalize
 * marker through `payload.kv`, and those records must live IN the
 * database they describe: a restore, branch, or clone of the database
 * stops carrying them otherwise, and eviction can lose the finalize
 * marker (the GHSA-xw9m binding semantics assume database durability).
 *
 * - `'none'` — no adapter configured: Payload's default, which is
 *   database-backed (the `payload-kv` collection / `payload_kv` table).
 * - `'database'` — `databaseKVAdapter()` (or equivalent) configured: it
 *   always ships the `kvCollection` it creates.
 * - `'external'` — any other adapter (Redis, …): the records live
 *   outside the database. Every kit command refuses this.
 */
export type KvAdapterKind = 'database' | 'external' | 'none';
export declare function kvAdapterOf(config: unknown): KvAdapterKind;
