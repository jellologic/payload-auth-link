// ===== codemods.d.ts =====
/**
 * Codemods (issue #53): rule-based rewrites for the call sites a
 * payload-auth app must migrate. Deliberately textual and scoped —
 * each rule matches an exact, idiomatic payload-auth pattern and
 * leaves anything it cannot fully rewrite untouched (reported in the
 * result), because a half-rewritten call site is worse than an
 * explicit one.
 */
export type CodemodRule = {
    readonly id: string;
    readonly description: string;
    /** Apply to one file's content; returns the new content (same string when untouched). */
    readonly apply: (content: string) => string;
};
export type CodemodFileResult = {
    readonly file: string;
    readonly changed: boolean;
    /** Rules that fired on this file. */
    readonly applied: readonly string[];
    /** Matches the rules could not fully rewrite (reported, never touched). */
    readonly manual: readonly string[];
};
export declare const codemodRules: readonly CodemodRule[];
/**
 * Report call sites the rules cannot fully rewrite — each one needs a
 * human decision (which session API call, which route handler shape).
 */
export declare function findManualSites(content: string): readonly string[];
/** Apply every rule to one file; reports what fired and what needs a human. */
export declare function applyCodemods(file: string, content: string): CodemodFileResult;

// ===== doctor/run-doctor.d.ts =====
import type { BehaviourChange, DoctorReport } from './types.js';
/**
 * The read-only doctor (issue #52): point it at a payload-auth app and
 * get the exact list of what breaks before touching anything. It loads
 * the app's config module, inspects the users collection and Better
 * Auth plugins, scans the source tree for call sites that must
 * migrate, and (with a database URL) reports row counts, unverified
 * users, and orphans. It never writes.
 */
export type DoctorOptions = {
    /** Absolute or cwd-relative path to the app's payload config. */
    readonly configPath: string;
    /** Source directory to scan. Defaults to the config's directory. */
    readonly srcDir?: string | undefined;
    /** Postgres connection string. Defaults to `DATABASE_URL`. */
    readonly databaseUrl?: string | undefined;
    /** Config loader, injectable for tests. Defaults to dynamic import. */
    readonly loadModule?: ((path: string) => Promise<unknown>) | undefined;
};
/**
 * Behaviour changes in payload-auth-link 0.7.0 that a payload-auth app
 * feels on day one. Static by nature; kept next to the doctor so the
 * report is one artifact.
 */
export declare const behaviourChanges: readonly BehaviourChange[];
/**
 * Detect payload-auth in the built config. The plugin stamps
 * `config.custom.hasBetterAuthPlugin = true` when it runs; older or
 * future builds that instead expose `pluginOptions` on the plugin
 * function are honored too.
 */
export declare function findPayloadAuthPlugin(config: unknown): {
    pluginOptions: Record<string, unknown>;
} | null;
/**
 * The Better Auth plugin ids the app declared. Reads them off
 * `betterAuthOptions.plugins` when a pluginOptions object is visible;
 * callers otherwise derive them from the built collections and the
 * config source (see {@link pluginsFromCollectionSlugs} and
 * {@link pluginsFromConfigSource}).
 */
export declare function betterAuthPluginIds(pluginOptions: Record<string, unknown>): string[];
/** The users collection slug the payload-auth app configured. */
export declare function usersSlugOf(pluginOptions: Record<string, unknown>): string;
/** Collection slugs of the built config. */
export declare function collectionSlugs(config: unknown): string[];
export declare function runDoctor(options: DoctorOptions): Promise<DoctorReport>;

// ===== doctor/types.d.ts =====
/**
 * The doctor's report shape (issue #52): read-only findings, every
 * one carrying a stable machine code so tests (and the guide) assert
 * exact blocker lists instead of prose.
 */
/** A finding the doctor is confident about: fix it before migrating. */
export type DoctorIssue = {
    /** Stable machine code — e.g. `better-auth-out-of-band`. */
    readonly code: string;
    /** Human detail: what was found, what closes it. */
    readonly detail: string;
};
/** One upstream dependency against its supported band. */
export type VersionFinding = {
    readonly name: string;
    /** `null` when no manifest resolved (exotic bundling). */
    readonly installed: string | null;
    /** The supported band, human-readable. */
    readonly band: string;
    readonly status: 'ok' | 'warning' | 'blocker' | 'unknown';
    readonly detail: string;
};
/** One Better Auth plugin's migration posture. */
export type PluginFinding = {
    /** The plugin's `id` (or the satellite package name). */
    readonly id: string;
    readonly status: 'supported' | 'auto-declarable' | 'unprobed' | 'unsupported' | 'removed-upstream';
    readonly detail: string;
};
/** The users collection, as the migration will meet it. */
export type UsersFinding = {
    /** Collection slug (payload-auth's `users` by default). */
    readonly slug: string;
    readonly roleField: {
        readonly name: string;
        readonly type: string;
        readonly hasMany: boolean;
        /** Declared select values, when the field is a select. */
        readonly values: readonly string[];
        /** Values that are not canonical lowercase slugs. */
        readonly nonCanonical: readonly string[];
    } | null;
    /** Payload's own email+password local strategy is active. */
    readonly localStrategy: boolean;
    /** Host fields colliding with the plugin's injected field names. */
    readonly collisions: readonly string[];
    readonly notes: readonly string[];
};
/** One source-scan category with the files that matched. */
export type SourceFinding = {
    readonly kind: 'payload-auth-import' | 'better-auth-accessor' | 'get-payload-auth' | 'ba-internals' | 'req-user-field';
    readonly files: readonly string[];
    /** For `req-user-field`: every field read off `req.user`. */
    readonly fields: readonly string[];
    readonly detail: string;
};
/** Database state, as far as a read-only connection can see. */
export type DatabaseFinding = {
    readonly tables: readonly {
        readonly name: string;
        readonly rows: number;
    }[];
    /** Users with `emailVerified = false` — they will not JIT-link. */
    readonly unverifiedUsers: number | null;
    readonly orphans: readonly {
        readonly kind: string;
        readonly count: number;
    }[];
};
/** Default-behaviour changes in payload-auth-link 0.7.0 that bite. */
export type BehaviourChange = {
    readonly title: string;
    readonly detail: string;
};
/** The full read-only doctor report (`--json` prints this object). */
export type DoctorReport = {
    readonly versions: readonly VersionFinding[];
    readonly plugins: readonly PluginFinding[];
    readonly users: UsersFinding | null;
    readonly source: readonly SourceFinding[];
    readonly database: DatabaseFinding | null;
    readonly behaviourChanges: readonly BehaviourChange[];
    readonly blockers: readonly DoctorIssue[];
    readonly warnings: readonly DoctorIssue[];
};

// ===== importer/apply.d.ts =====
import type { Payload } from 'payload';
import { type ImportManifest, type ImportResult, type PlanRow } from './types.js';
/**
 * Apply side (issue #50): copy payload-auth rows into the
 * payload-auth-link collections through the Payload Local API.
 *
 * - Password hashes, OAuth tokens, API-key hashes, and jwks carry over
 *   VERBATIM (payload-auth used Better Auth's own hashing).
 * - `linkable` + `linkableEmail` are granted from VERIFIED emails only;
 *   unverified users are reported by the plan, never linked.
 * - Sessions default to `carry` so nobody is logged out: the carried
 *   rows keep their tokens, and with the same `advanced.cookiePrefix`
 *   the existing cookies keep working.
 * - Idempotent and resumable: every target row carries a deterministic
 *   `baId` derived from the source id (`acct-<source id>`), so a
 *   re-run after a crash finds its own earlier rows and skips them.
 * - Every created row and every field write lands in the manifest, so
 *   `rollback` is exact and the source tables stay untouched.
 */
export type ApplyOptions = {
    readonly payload: Payload;
    readonly sourceConnectionString: string;
    readonly apply: boolean;
    readonly sessions: 'carry' | 'drop';
    readonly manifest: ImportManifest | null;
    /** Written (merged) on `apply`; required when `apply` is true. */
    readonly manifestPath: string;
    readonly sourceCounts: Readonly<Record<string, number>>;
    readonly sourceDatabase: string;
    /** Target collection slugs; defaults must match the engine wiring. */
    readonly slugs?: Readonly<Record<string, string>> | undefined;
    /** Injectable clock for deterministic `linkedAt` stamps. */
    readonly now?: () => Date;
    readonly log?: (message: string) => void;
};
export declare function applyImport(options: ApplyOptions): Promise<ImportResult>;
/** Collapse per-row plan entries into one line per model (dry-run print). */
export declare function summarizePlanRows(rows: readonly PlanRow[]): PlanRow[];

// ===== importer/lifecycle.d.ts =====
import type { Payload } from 'payload';
import type { ImportManifest } from './types.js';
/**
 * Manifest lifecycle (issue #50): `verify` checks both sides, `rollback`
 * undoes exactly what the manifest records, and `finalize` marks the
 * manifest so the source tables may be dropped — by the operator, as a
 * separate later step.
 */
export declare const defaultManifestPath = "payload-auth-link-migration.json";
export declare function readManifest(path: string): ImportManifest | null;
export declare function writeManifest(manifest: ImportManifest, path: string): void;
export type VerifyIssue = {
    readonly code: string;
    readonly detail: string;
};
export type VerifyResult = {
    readonly ok: boolean;
    readonly issues: readonly VerifyIssue[];
    readonly checks: readonly {
        readonly name: string;
        readonly passed: boolean;
    }[];
};
/**
 * Verify (issue #50): per-model counts match, every source user has
 * exactly one target row, every credential account resolves to its
 * user, every API key carries its hash with a resolvable owner, and
 * every member/invitation references an existing org and user. Exits
 * non-zero on any mismatch (the caller maps `ok` to the exit code).
 */
export declare function verifyImport(options: {
    readonly payload: Payload;
    readonly sourceConnectionString: string;
    readonly slugs?: Readonly<Record<string, string>> | undefined;
}): Promise<VerifyResult>;
/**
 * Rollback (issue #50): delete the created rows in reverse order and
 * restore every captured field value. The manifest records exactly
 * what to undo — nothing else is touched, and the source tables were
 * never modified.
 */
export declare function rollbackImport(options: {
    readonly payload: Payload;
    readonly manifest: ImportManifest;
    readonly log?: (message: string) => void;
}): Promise<{
    deleted: number;
    restored: number;
}>;
/** Finalize (issue #50): the point of no return — mark the manifest. */
export declare function finalizeManifest(manifest: ImportManifest, path: string): ImportManifest;

// ===== importer/run-import.d.ts =====
import { type VerifyResult } from './lifecycle.js';
import { type ImportManifest, type ImportPlan, type ImportResult } from './types.js';
/**
 * The import commands (issue #50): `from-payload-auth` (dry-run by
 * default, `--apply` writes), `verify`, `rollback`, `finalize`.
 *
 * Dry run prints planned row counts per model, collisions, and
 * anything unmappable; unmappable data aborts `--apply` before any
 * write. The manifest records every created row and field change;
 * source tables are never modified — `finalize` only marks the
 * manifest, and dropping them stays the operator's later step.
 */
export type ImportCommandOptions = {
    readonly configPath: string;
    readonly databaseUrl: string;
    readonly apply: boolean;
    readonly sessions: 'carry' | 'drop';
    readonly manifestPath?: string | undefined;
    readonly slugsJson?: string | undefined;
    readonly json?: boolean;
    readonly log?: (message: string) => void;
    readonly loadPayload?: (configPath: string) => Promise<import('payload').Payload>;
};
/** The `from-payload-auth` command: dry-run plan, or `--apply` + manifest. */
export declare function runFromPayloadAuth(options: ImportCommandOptions): Promise<{
    exitCode: number;
    result?: ImportResult;
    plan: ImportPlan;
    unverified: readonly string[];
}>;
/** The `verify` command. */
export declare function runVerify(options: ImportCommandOptions): Promise<VerifyResult>;
/** The `rollback` command. */
export declare function runRollback(options: ImportCommandOptions): Promise<{
    deleted: number;
    restored: number;
}>;
/** The `finalize` command. */
export declare function runFinalize(options: ImportCommandOptions): Promise<ImportManifest>;

import { Client } from 'pg';

    readonly providerId: string;

    readonly accessToken: string | null;
    readonly refreshToken: string | null;
    readonly idToken: string | null;

    readonly refreshTokenExpiresAt: Date | null;

// ===== importer/types.d.ts =====
/**
 * Importer manifest (issue #50): every row created and every field
 * changed, so `rollback` is exact. The manifest is the run's contract —
 * `verify`, `rollback`, and `finalize` all read it.
 */
export type CreatedRow = {
    /** Target collection slug. */
    readonly collection: string;
    /** Payload document id of the created row. */
    readonly id: string | number;
    /** The source identity this row was created for (dedupe on resume). */
    readonly sourceKey: string;
};
export type FieldWrite = {
    readonly collection: string;
    readonly id: string | number;
    readonly field: string;
    /** The prior value, JSON-serialized, restored by rollback. */
    readonly before: string | null;
};
export type ImportManifest = {
    readonly version: 1;
    readonly createdAt: string;
    readonly sourceDatabase: string;
    readonly options: {
        readonly sessions: 'carry' | 'drop';
    };
    readonly created: CreatedRow[];
    readonly fieldWrites: FieldWrite[];
    readonly finalized: boolean;
    /** Counts per source table at plan time (the rollback baseline). */
    readonly sourceCounts: Readonly<Record<string, number>>;
    /** Cumulative plan totals (rows created + skipped across resumes). */
    readonly totals: Readonly<Record<string, number>>;
};
/** Per-model plan row for the dry run. */
export type PlanRow = {
    readonly model: string;
    readonly sourceTable: string;
    readonly targetCollection: string;
    readonly rows: number;
    readonly skipped: number;
    readonly unmappable: readonly string[];
};
/** Why a row cannot be imported — the run aborts rather than skips. */
export type Unmappable = {
    readonly model: string;
    readonly key: string;
    readonly reason: string;
};
export type ImportPlan = {
    readonly rows: readonly PlanRow[];
    readonly unmappable: readonly Unmappable[];
    readonly sessions: 'carry' | 'drop';
    readonly warnings: readonly string[];
};
export type ImportResult = {
    readonly created: number;
    readonly skipped: number;
    readonly fieldWrites: number;
    readonly plan: ImportPlan;
};
/** The models the importer copies. */
export type TargetModel = 'user' | 'session' | 'account' | 'verification' | 'apikey' | 'jwks' | 'organization' | 'member' | 'invitation';
/** Target slugs with the known models always present (index-safe). */
export type TargetSlugMap = Record<TargetModel, string> & Record<string, string>;
/** Default target slugs; must match the app's engine wiring. */
export declare const defaultTargetSlugs: TargetSlugMap;
