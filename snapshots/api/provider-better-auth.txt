// ===== adapter.d.ts =====
import type { BetterAuthOptions } from 'better-auth';
import { type AdapterFactory } from 'better-auth/adapters';
import type { Payload } from 'payload';
import { type BetterAuthCollectionSlugs, type BetterAuthCustomField, type BetterAuthCustomModel, type BetterAuthModelKey, type BetterAuthSchemaMap } from './collections.js';
/** Database adapter wiring: Payload access, slugs, read depth. */
export type PayloadDatabaseAdapterDeps = {
    /** Resolves the Payload instance (cached `getPayload` in Next.js). */
    readonly getPayload: () => Promise<Payload>;
    /** Collection slugs per model. Defaults to {@link defaultBetterAuthSlugs}. */
    readonly slugs?: Partial<BetterAuthCollectionSlugs> | undefined;
    /**
     * Adopt existing collections for these models (issue #49): the
     * adapter serves the model from the host slug, maps renamed fields,
     * and — with `idStrategy: 'payload'` — uses the Payload document id
     * as the Better Auth row id. No `ba-*` collection exists for an
     * adopted model.
     */
    readonly schemaMap?: BetterAuthSchemaMap | undefined;
    /**
     * `'payload'` (adopted models only) makes the Payload document id
     * the Better Auth row id, so ids never move. `'ba'` (the default)
     * keeps the provider id in the dedicated `baId` column of the
     * collections the engine creates.
     */
    readonly idStrategy?: 'ba' | 'payload' | undefined;
    /** Payload query depth for adapter reads. Rows are flat; defaults to 0. */
    readonly depth?: number | undefined;
    /** Declared custom fields on core models; must match the BA schema exactly. */
    readonly customFields?: Partial<Record<BetterAuthModelKey, readonly BetterAuthCustomField[]>> | undefined;
    /** Declared custom plugin models; must match the BA schema exactly. */
    readonly customModels?: readonly BetterAuthCustomModel[] | undefined;
};
/**
 * Better Auth database adapter over the Payload Local API. System-level
 * persistence (like the plugin repository): reads and writes bypass access
 * control with `overrideAccess`, and the collections default to deny-all
 * API access.
 *
 * The returned factory is passed as Better Auth's `database` option; the
 * factory validates the Better Auth schema against the Payload mapping
 * when the instance is created, failing fast on renames and unmapped
 * fields. Transactions run sequentially (Payload commits per operation).
 */
export declare function createPayloadDatabaseAdapter(deps: PayloadDatabaseAdapterDeps): AdapterFactory<BetterAuthOptions>;

// ===== collections.d.ts =====
import type { CollectionConfig } from 'payload';
/** Access control for the Better Auth system collections. */
export type BetterAuthCollectionsAccess = NonNullable<CollectionConfig['access']>;
/** Canonical Better Auth models persisted in Payload. */
export type BetterAuthModelKey = 'user' | 'session' | 'account' | 'verification' | 'rateLimit';
/** Payload collection slugs backing each Better Auth model. */
export type BetterAuthCollectionSlugs = Record<BetterAuthModelKey, string>;
/** Default `ba-*` slugs; override per model via options. */
export declare const defaultBetterAuthSlugs: BetterAuthCollectionSlugs;
/** Payload column types a custom field can map to. */
export type BetterAuthCustomFieldType = 'text' | 'number' | 'boolean' | 'date' | 'json';
/**
 * A builder-declared custom field: an `additionalFields` entry (or plugin
 * field) on a core model, or a field of a custom model. The declaration
 * must match Better Auth's schema exactly — the adapter fail-fasts both
 * undeclared schema fields and declared-but-absent names.
 */
export type BetterAuthCustomField = {
    readonly name: string;
    /** Payload column type; match Better Auth's attribute type. */
    readonly type: BetterAuthCustomFieldType;
    readonly required?: boolean | undefined;
    /** Unique + indexed (text/number/date only). */
    readonly unique?: boolean | undefined;
};
/**
 * A builder-declared custom model: a plugin model (e.g. the organization
 * plugin's `organization`, `member`, `invitation`) the adapter should
 * serve. Gets its own `ba-*` collection with a `baId` id column.
 */
export type BetterAuthCustomModel = {
    /** Better Auth model name, exactly as the plugin registers it. */
    readonly model: string;
    /** Collection slug. Defaults to `ba-${model}`. */
    readonly slug?: string | undefined;
    /** Admin label. Defaults to `Better Auth ${model}`. */
    readonly label?: string | undefined;
    readonly fields: readonly BetterAuthCustomField[];
};
/**
 * One adopted model (issue #49): the engine serves a Better Auth model
 * from an EXISTING Payload collection instead of creating its own
 * `ba-*` collection — the "adopt, don't replace" path for apps whose
 * auth rows already live in Payload (payload-auth layout).
 */
export type BetterAuthAdoptedModel = {
    /** The host collection slug that backs the model. */
    readonly slug: string;
    /**
     * Better Auth field name → host Payload field name, for fields the
     * host renamed. Most payload-auth layouts rename only the
     * relationship fields (`userId` → `user`); everything unmapped keeps
     * its Better Auth name. Renamed fields read and write as ids: the
     * adapter reads them at depth 0 and writes the id the host's
     * relationship expects, coercing to the Payload id type.
     */
    readonly fields?: Readonly<Record<string, string>> | undefined;
};
/**
 * Adopted models by Better Auth model name — core keys (`user`,
 * `session`, `account`, `verification`, `rateLimit`) and plugin models
 * (`organization`, `member`, `invitation`, `apikey`, `jwks`, …) alike.
 */
export type BetterAuthSchemaMap = Readonly<Record<string, BetterAuthAdoptedModel>>;
/** Slug overrides and access for {@link createBetterAuthCollections}. */
export type BetterAuthCollectionsOptions = {
    /** Override any default slug. Unmentioned models keep their defaults. */
    readonly slugs?: Partial<BetterAuthCollectionSlugs> | undefined;
    /**
     * Adopt existing collections for these models: the engine creates no
     * collection for them, forces no access, and serves the model from
     * the host slug (issue #49). The host keeps its access, hooks,
     * versions, and admin config; its extra fields are invisible to
     * Better Auth and preserved because the adapter writes partial
     * updates only.
     */
    readonly schemaMap?: BetterAuthSchemaMap | undefined;
    /**
     * Access control for the system collections. Defaults to deny-all: these
     * tables hold credential material (hashes, tokens) and the adapter writes
     * with `overrideAccess`, so no API access is needed. Builders may open
     * `read` for operations debugging. Never applied to adopted models.
     */
    readonly access?: BetterAuthCollectionsAccess | undefined;
    /** Custom fields appended to core-model collections. */
    readonly customFields?: Partial<Record<BetterAuthModelKey, readonly BetterAuthCustomField[]>> | undefined;
    /** Custom plugin-model collections. */
    readonly customModels?: readonly BetterAuthCustomModel[] | undefined;
};
/**
 * Canonical Better Auth core fields per model (mirrors `get-tables`; `id`
 * is implicit). Single source of truth shared by the collections (which
 * must not collide with it) and the adapter (which allowlists it).
 */
export declare const betterAuthCoreFields: Record<BetterAuthModelKey, readonly string[]>;
/** Default slug for a custom model collection. */
export declare function defaultCustomSlug(model: string): string;
/** Validated collection slug for a custom model declaration. */
export declare function customModelSlug(entry: BetterAuthCustomModel): string;
/**
 * Payload collections backing the five Better Auth core models plus any
 * declared custom fields and custom models. Spread the result into the
 * Payload config `collections` array; the engine wires the adapter
 * against the same slugs.
 *
 * Field lists mirror Better Auth's canonical schema; the adapter rejects
 * renamed models/fields, undeclared extras, and declared-but-absent names
 * fail-fast at startup.
 */
export declare function createBetterAuthCollections(options?: BetterAuthCollectionsOptions): CollectionConfig[];
/**
 * Validate an adoption map at wiring time (issue #49): models must be
 * known (core or declared custom), entries must name a slug, and field
 * maps must not rename reserved names. The host collection's own
 * fields are validated once the Payload instance exists (the adapter's
 * first use) — wiring time has no collections to compare against.
 */
export declare function validateSchemaMap(schemaMap: BetterAuthSchemaMap, known: {
    readonly coreModels: readonly string[];
    readonly customModels: readonly string[];
}): void;

// ===== cookies.d.ts =====
import { type CookieOptions, type SecureCookieEnvironment } from '@payload-auth-link/core';
import type { BetterAuthOptions } from 'better-auth';
/** The Better Auth options that influence cookie names. */
export type BetterAuthCookieOptions = Pick<BetterAuthOptions, 'advanced' | 'baseURL' | 'session'>;
/**
 * Cookies to expire on logout that do not indicate a live session (the
 * port's `logoutCookieNames`): today only the admin plugin's
 * `admin_session`. Expired unconditionally — when the admin plugin is
 * absent the cookie simply never exists and expiring it is a no-op,
 * while leaving a present one behind would keep the impersonating
 * operator's resumable session token in the browser after logout.
 */
export declare function getBetterAuthLogoutCookieNames(options: BetterAuthCookieOptions, environment?: SecureCookieEnvironment): readonly string[];
/**
 * Session cookie names for a Better Auth instance. See
 * `sessionCookieEntries` for the mirroring rule; the cookie-suite drift
 * test asserts these against BA's real cookie getter.
 */
export declare function getBetterAuthSessionCookieNames(options: BetterAuthCookieOptions, environment?: SecureCookieEnvironment): readonly string[];
/**
 * Logout clearing lookup: the live overwrite attributes per session
 * cookie. Attribute precedence mirrors BA: per-cookie
 * `advanced.cookies.*.attributes` win, then
 * `advanced.defaultCookieAttributes`, then the secure default above.
 * Unknown names yield `undefined` so the caller falls back to
 * `expiringCookieOptions`.
 *
 * Limitation: `crossSubDomainCookies` derives the cookie domain from the
 * base URL inside BA; that derivation is not mirrored here, so
 * cross-subdomain deployments should sign out through BA's own sign-out
 * for full cookie clearing. Server-side revocation still applies, so any
 * surviving cookie is dead — the plugin fails closed on it.
 */
export declare function getBetterAuthSessionCookieClearOptions(options: BetterAuthCookieOptions, environment?: SecureCookieEnvironment): (name: string) => CookieOptions | undefined;

// ===== derive-schema.d.ts =====
/**
 * Derive the engine's `customFields` / `customModels` declarations
 * from Better Auth's own schema (issue #17): `getAuthTables(options)`
 * is the same source the adapter validates against, so a derived
 * declaration cannot disagree with what BA will ask for. Hand
 * declarations stay available (and win on conflicts when
 * `autoDeclare` merges) — derivation is for builders who compose
 * many plugins and do not want to hand-copy field lists per BA bump.
 */
import { getAuthTables } from 'better-auth/db';
import { type BetterAuthCustomField, type BetterAuthCustomModel, type BetterAuthModelKey } from './collections.js';
type Tables = ReturnType<typeof getAuthTables>;
/** Fields Payload owns end to end (timestamps, ids); never declared. */
export declare const payloadOwnedFields: ReadonlySet<string>;
/** Whether a BA model key is one of the adapter's core models. */
export declare function isCoreModel(model: string): model is BetterAuthModelKey;
/**
 * Pure derivation over a `getAuthTables` result: core models keep
 * their plugin-added fields, everything else becomes a custom model.
 * Model or field renames (`modelName` / `fieldName` overrides) throw:
 * the adapter serves canonical BA names, and silently declaring a
 * renamed table under its key would produce a collection BA never
 * queries — declare those by hand.
 */
export declare function declarationsFromTables(tables: Tables): {
    customFields: Partial<Record<BetterAuthModelKey, BetterAuthCustomField[]>>;
    customModels: BetterAuthCustomModel[];
};
/**
 * Derive declarations for a Better Auth options object — the shape
 * the builder passes to `betterAuth`/`createBetterAuthEngine`, whose
 * plugin list determines the schema.
 */
export declare function deriveBetterAuthSchema(options: unknown): {
    customFields: Partial<Record<BetterAuthModelKey, BetterAuthCustomField[]>>;
    customModels: BetterAuthCustomModel[];
};
export {};

// ===== engine.d.ts =====
import { type IdentityProviderPort, type LoggerPort, type VersionRange } from '@payload-auth-link/core';
import { type BetterAuthOptions, type BetterAuthPlugin } from 'better-auth';
import type { CollectionConfig, Payload } from 'payload';
import { type BetterAuthCollectionsAccess, type BetterAuthCollectionSlugs, type BetterAuthCustomField, type BetterAuthCustomModel, type BetterAuthModelKey, type BetterAuthSchemaMap } from './collections.js';
/** Engine wiring: Payload access plus the builder's Better Auth options. */
export type BetterAuthEngineInput<TOptions extends Omit<BetterAuthOptions, 'database'>> = {
    /** Resolves the Payload instance (cached `getPayload` in Next.js). */
    readonly getPayload: () => Promise<Payload>;
    /** Override any default Better Auth collection slug. */
    readonly slugs?: Partial<BetterAuthCollectionSlugs> | undefined;
    /**
     * Adopt existing collections for these Better Auth models
     * (issue #49): the engine creates no `ba-*` collection for an
     * adopted model and serves it from the host slug instead — the
     * payload-auth layout boots with zero row moves. The host keeps its
     * access, hooks, versions, and admin config; its extra fields are
     * invisible to Better Auth and survive every write (the adapter
     * writes partial updates only).
     */
    readonly schemaMap?: BetterAuthSchemaMap | undefined;
    /**
     * `'payload'` (adopted models only) makes the Payload document id
     * the Better Auth row id, so existing ids never move:
     * `schemaMap: { user: { slug: 'users' } }, idStrategy: 'payload'`.
     * `'ba'` (default) keeps the provider id in a dedicated column on
     * the collections the engine creates.
     */
    readonly idStrategy?: 'ba' | 'payload' | undefined;
    /**
     * Coexistence window (issue #51): legacy payload-auth session
     * cookie names honored READ-ONLY until the ISO `until` date — the
     * lookup re-presents them under the current names, minting always
     * uses the current names, and after the window they are refused.
     * Exposed on the engine as `legacySessionCookieNames` for the
     * cookie helpers and Edge middleware.
     */
    readonly legacySessions?: {
        readonly until: string;
        readonly cookieNames: readonly string[];
    } | undefined;
    /** Custom fields on core models (`additionalFields`, plugin fields). */
    readonly customFields?: Partial<Record<BetterAuthModelKey, readonly BetterAuthCustomField[]>> | undefined;
    /** Custom plugin models (e.g. the organization plugin's tables). */
    readonly customModels?: readonly BetterAuthCustomModel[] | undefined;
    /**
     * Derive `customFields`/`customModels` from Better Auth's own
     * `getAuthTables(options)` (issue #17) and merge them UNDER the
     * builder's explicit declarations — a hand declaration always wins
     * over a derived one, and the derived remainder fills the gaps.
     * Strict hand-declared mode is unchanged when this is absent.
     */
    readonly autoDeclare?: boolean | undefined;
    /** Payload query depth for adapter reads. Defaults to 0. */
    readonly depth?: number | undefined;
    /** Access control for the system collections. Defaults to deny-all. */
    readonly collectionsAccess?: BetterAuthCollectionsAccess | undefined;
    /**
     * Structured logger for wiring-time risky-linking warnings (CAP-12).
     * Defaults to silent (`nullLogger`) — pass the same logger as the
     * plugin to see them. Warnings name the offending
     * `account.accountLinking` setting; they never go to `console.*`.
     */
    readonly logger?: LoggerPort | undefined;
    /**
     * Turn every risky-linking warning into a wiring-time
     * `InvalidConfigError` (CAP-12). Risky means a Better Auth setting
     * that can let an unverified email claim link accounts inside the
     * provider: `account.accountLinking.trustedProviders` naming a
     * provider without `requireEmailVerification`,
     * `allowDifferentEmails: true`, `requireLocalEmailVerified: false`,
     * or `updateUserInfoOnLink: true`. The plugin's verified-email gate
     * still guards Payload linking either way; this option makes the
     * provider-side loosening fail fast instead of merely loud.
     * Defaults to `false` (warn only).
     */
    readonly strictLinking?: boolean | undefined;
    /**
     * The builder's Better Auth options, passed through verbatim except for
     * `database` (wired to Payload here, so it is omitted from this type),
     * `emailAndPassword` (see below), and
     * `account.accountLinking.updateUserInfoOnLink`, which the engine
     * forces off so a link can never rewrite `email`/`emailVerified`
     * (CAP-12; see `strictLinking`).
     *
     * Framework cookie plugins (`nextCookies()`, `tanstackStartCookies()`,
     * …) belong in `options.plugins`, exactly as Better Auth documents
     * them — the engine never touches the plugin list, so any present or
     * future cookie plugin works without the engine knowing about it.
     */
    readonly options: TOptions;
};
/** Composed engine: instance, port, collections, slugs, cookie names. */
export type BetterAuthEngine<AuthType> = {
    /** The composed Better Auth instance; mount it with the Next adapter. */
    readonly auth: AuthType;
    /** The instance behind `IdentityProviderPort`; register it with the plugin. */
    readonly provider: IdentityProviderPort;
    /** Payload collections for the Better Auth models; spread into the config. */
    readonly collections: CollectionConfig[];
    /** Resolved slugs (defaults plus overrides). */
    readonly slugs: BetterAuthCollectionSlugs;
    /** The adoption map the engine serves (issue #49). */
    readonly schemaMap: BetterAuthSchemaMap;
    /** Legacy cookie names honored during the coexistence window (#51). */
    readonly legacySessionCookieNames: readonly string[];
    /** Session cookie names for middleware fast-paths and logout clearing. */
    readonly sessionCookieNames: readonly string[];
};
/**
 * Compose a Better Auth instance on Payload storage: the Payload database
 * adapter, secure credential defaults, the `IdentityProviderPort`, and
 * the Payload collections.
 *
 * The return type is inferred so Better Auth's own option-driven typing
 * (plugin endpoints, `$Infer` session types) flows through untouched.
 * Builder options (including `plugins`) pass through verbatim — the
 * engine appends nothing, so framework cookie plugins are declared by
 * the builder exactly as Better Auth documents them.
 *
 * Secure defaults: when `emailAndPassword` is present but silent on
 * `requireEmailVerification`, verification is required — builders must
 * explicitly opt out (the example does, and documents why). The user
 * bridge provisions Payload users lazily and idempotently at session
 * verification (see the plugin session service), so no signup
 * ordering can strand a provider user. The engine does compose
 * `databaseHooks` for CAP-14 lifecycle translation (user update /
 * delete, account delete → core `IdentityEvent`s on the port's
 * `onIdentityEvent`), preserving any builder-supplied hooks.
 */
/**
 * better-auth versions this release validated (Phase-1 bisect): the
 * type-level floor is 1.7.6 (BA's adapter `modelKey` types), and the
 * band admits only the validated 1.7 line — BA satellites force
 * minors (B12), so an unvalidated minor must fail here, typed and
 * before any DB write, never as a missing adapter method
 * mid-request. Widened only by the compatibility matrix, never by
 * optimism.
 */
export declare const supportedBetterAuthBand: readonly VersionRange[];

    readonly api: {
        readonly getSession: (input: BetterAuthSessionQuery) => Promise<unknown>;
