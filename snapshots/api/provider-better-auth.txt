// ===== account-state.d.ts =====
import { APIError } from 'better-auth/api';
import type { BetterAuthPlugin } from 'better-auth';
import { type LoggerPort } from '@payload-auth-link/core';
/**
 * Account-state enforcement on the Better Auth side (issues
 * GHSA-62xf-qq8f-x6w9 and GHSA-xxff-hh6q-c99p).
 *
 * Suspension (`suspendedUntil`, the plugin's CAP-7 field) and Better
 * Auth admin bans (`banned` + `banExpires`) were enforced only where
 * payload-auth-link resolves a session — the Payload strategy, the
 * server API, API keys. Better Auth's own endpoints kept working with
 * the same cookie: a suspended user could sign in fresh, read
 * `get-session`, mint API keys, update their profile; a banned user
 * kept every live session (Better Auth itself refuses bans only at
 * new sign-ins, and its revocation path was broken on adopted role
 * fields).
 *
 * This plugin closes that at the engine boundary:
 * - `hooks.before` on every request: when a session resolves, the user
 *   row is read fresh and an ACTIVE ban or suspension refuses the
 *   request with 403 — so `get-session` un-authenticates, and every
 *   authenticated provider endpoint (api keys, organization
 *   management, update-user, …) refuses.
 * - `databaseHooks.session.create.before`: the same check refuses the
 *   session at creation, so a suspended or banned user cannot sign in
 *   fresh.
 *
 * The user row is read FRESH per request (never from a cached
 * session): the state can change between requests, which is the point.
 * `suspendedUntil` is read from the configured field (the plugin's
 * CAP-7 field name); `banned`/`banExpires` are Better Auth's own admin
 * columns.
 */
export type AccountStateOptions = {
    /** The CAP-7 suspension field on the user row. Default: `suspendedUntil`. */
    readonly suspendedUntilField?: string | undefined;
    /**
     * Read the user row FRESH through Payload (issue GHSA-jvpf-3fhq-f5ph):
     * Better Auth's internal adapter returns only BA-schema fields, and
     * the CAP-7 `suspendedUntil` field is not declared there — reading
     * through the internal adapter always sees it as undefined. When
     * provided, the endpoint-boundary check reads through this instead
     * (the engine passes a lookup honoring `schemaMap.user`,
     * `idStrategy`, and `suspendedUntilField`). Optional only for
     * payloads without the plugin's users fields.
     */
    readonly readUserRow?: ((userId: string) => Promise<unknown>) | undefined;
    readonly logger?: LoggerPort | undefined;
};
export declare function activeBanOf(user: Record<string, unknown>, now: Date): boolean;
export declare function activeSuspensionOf(user: Record<string, unknown>, field: string, now: Date): boolean;
/** One fresh user-row read + the two state checks. Returns null when allowed. */
export declare function accountStateRefusal(userId: string, field: string, readUserRow: (id: string) => Promise<unknown>, logger: LoggerPort): Promise<APIError | null>;
export declare function createAccountStatePlugin(options?: AccountStateOptions): BetterAuthPlugin;

/** Database adapter wiring: Payload access, slugs, read depth. */
export type PayloadDatabaseAdapterDeps = {
    /** Resolves the Payload instance (cached `getPayload` in Next.js). */
    readonly getPayload: () => Promise<Payload>;
    /** Collection slugs per model. Defaults to {@link defaultBetterAuthSlugs}. */
    readonly slugs?: Partial<BetterAuthCollectionSlugs> | undefined;
    /**
     * Adopt existing collections for these models (issue #49): the
     * adapter serves the model from the host slug, maps renamed fields,
     * and — with `idStrategy: 'payload'` — uses the Payload document id
     * as the Better Auth row id. No `ba-*` collection exists for an
     * adopted model.
     */
    readonly schemaMap?: BetterAuthSchemaMap | undefined;
    /**
     * `'payload'` (adopted models only) makes the Payload document id
     * the Better Auth row id, so ids never move. `'ba'` (the default)
     * keeps the provider id in the dedicated `baId` column of the
     * collections the engine creates.
     */
    readonly idStrategy?: 'ba' | 'payload' | undefined;
    /** Payload query depth for adapter reads. Rows are flat; defaults to 0. */
    readonly depth?: number | undefined;
    /** Declared custom fields on core models; must match the BA schema exactly. */
    readonly customFields?: Partial<Record<BetterAuthModelKey, readonly BetterAuthCustomField[]>> | undefined;
    /** Declared custom plugin models; must match the BA schema exactly. */
    readonly customModels?: readonly BetterAuthCustomModel[] | undefined;
};
/**
 * Better Auth database adapter over the Payload Local API. System-level
 * persistence (like the plugin repository): reads and writes bypass access
 * control with `overrideAccess`, and the collections default to deny-all
 * API access.
 *
 * The returned factory is passed as Better Auth's `database` option; the
 * factory validates the Better Auth schema against the Payload mapping
 * when the instance is created, failing fast on renames and unmapped
 * fields. Transactions run sequentially (Payload commits per operation).
 *
 * Query semantics match Better Auth's reference `memoryAdapter` (pinned
 * by `adapter-differential.test.ts`) with these documented differences:
 *
 * - **String sort collation is the database's, not `localeCompare`.**
 *   `findMany({ sortBy })` delegates ordering to Payload, which exposes
 *   no portable collation option: SQLite and MongoDB order by binary
 *   (code point: `'A' < 'B' < 'a' < 'b'`), Postgres by the column's
 *   collation (often locale-aware). The memory adapter orders with
 *   `String.prototype.localeCompare` (`'a' < 'A' < 'b' < 'B'`).
 *   Normalising in memory would break `limit`/`offset` pagination (the
 *   window is cut by the database), so the difference is documented
 *   rather than papered over. Better Auth core issues no string-sorted
 *   paged reads; numbers, dates, and same-case strings agree.
 * - **`null` ordering** follows the database (SQLite/MongoDB: first in
 *   ascending order, like the memory adapter; Postgres: last).
 * - **String range operators** (`lt`/`lte`/`gt`/`gte` on text) compare
 *   in the same database collation as sorting; the memory adapter
 *   compares JS strings. Same-case ASCII strings agree everywhere.
 * - **Unordered reads** (no `sortBy`) return rows in Payload's default
 *   order, not insertion order; callers must not rely on either.
 * - **Pattern operators** (`contains`, `starts_with`, `ends_with`),
 *   case-insensitive `ne`/`in`/`not_in`, and a case-insensitive `eq`
 *   combined with an `OR` connector throw `UnsupportedOperatorError`
 *   rather than risk a silently wrong translation.
 *
 * Agreeing by construction (pinned by the same suite, #110): `ne` and
 * `not_in` keep `NULL` rows (the `not_in` translation adds an
 * `IS NULL` branch), and case-insensitive `eq` decides in JS, so it is
 * Unicode-aware on SQLite, whose `LIKE` folds ASCII only.
 */
export declare function createPayloadDatabaseAdapter(deps: PayloadDatabaseAdapterDeps): AdapterFactory<BetterAuthOptions>;

    readonly label?: string | undefined;

/**
 * One adopted model (issue #49): the engine serves a Better Auth model

    /** The host collection slug that backs the model. */
    readonly slug: string;

    readonly fields?: Readonly<Record<string, string>> | undefined;

/**
 * Adopted models by Better Auth model name — core keys (`user`,

/** Slug overrides and access for {@link createBetterAuthCollections}. */
export type BetterAuthCollectionsOptions = {
    /** Override any default slug. Unmentioned models keep their defaults. */
    readonly slugs?: Partial<BetterAuthCollectionSlugs> | undefined;
    /**
     * Adopt existing collections for these models: the engine creates no
     * collection for them, forces no access, and serves the model from

    readonly schemaMap?: BetterAuthSchemaMap | undefined;

     * Access control for the system collections. Defaults to deny-all: these

     */
    readonly access?: BetterAuthCollectionsAccess | undefined;
    /** Custom fields appended to core-model collections. */
    readonly customFields?: Partial<Record<BetterAuthModelKey, readonly BetterAuthCustomField[]>> | undefined;
    /** Custom plugin-model collections. */
    readonly customModels?: readonly BetterAuthCustomModel[] | undefined;
};

export declare function customModelSlug(entry: BetterAuthCustomModel): string;
/**

// ===== cookies.d.ts =====
import { type CookieOptions, type SecureCookieEnvironment } from '@payload-auth-link/core';
import type { BetterAuthOptions } from 'better-auth';
/** The Better Auth options that influence cookie names. */
export type BetterAuthCookieOptions = Pick<BetterAuthOptions, 'advanced' | 'baseURL' | 'session'>;
/**
 * Cookies to expire on logout that do not indicate a live session (the
 * port's `logoutCookieNames`):
 *
 * - the admin plugin's `admin_session` (the impersonating operator's
 *   resumable session token);
 * - `dont_remember` (sign-in without "remember me");
 * - `account_data` (the account cookie cache, when
 *   `account.storeAccountCookie` is on);
 * - the cookie-cache `session_data`, even when the cache is disabled
 *   now — it may have been enabled when the cookie was set;
 * - the `__Secure-` / plain counterpart of every session and logout
 *   name, so a deployment that switched between http and https (or
 *   toggled `useSecureCookies`) leaves no residue under the other
 *   name.
 *
 * Expired unconditionally — a cookie that was never set makes its
 * clearing write a no-op. Chunks of these cookies (`<name>.0` …) are
 * per-request and enumerated from the presented Cookie header by the
 * port's `presentedLogoutCookieNames`.
 */
export declare function getBetterAuthLogoutCookieNames(options: BetterAuthCookieOptions, environment?: SecureCookieEnvironment): readonly string[];
/**
 * Session cookie names for a Better Auth instance. See
 * `sessionCookieEntries` for the mirroring rule; the cookie-suite drift
 * test asserts these against BA's real cookie getter.
 */
export declare function getBetterAuthSessionCookieNames(options: BetterAuthCookieOptions, environment?: SecureCookieEnvironment): readonly string[];
/**
 * Logout clearing lookup: the live overwrite attributes per session
 * cookie. Attribute precedence mirrors BA: per-cookie
 * `advanced.cookies.*.attributes` win, then
 * `advanced.defaultCookieAttributes`, then the secure default above.
 * Unknown names yield `undefined` so the caller falls back to
 * `expiringCookieOptions`.
 *
 * `crossSubDomainCookies` is mirrored like BA's `createCookieGetter`:
 * when enabled, every cookie is written with `Domain` set to
 * `crossSubDomainCookies.domain`, or the static `baseURL`'s hostname —
 * underneath `defaultCookieAttributes` and per-cookie attributes, which
 * still win. Without the matching `Domain` a clearing write misses the
 * cookie, and the browser keeps the cookie-cache `session_data` that
 * BA's own `/get-session` keeps answering from until it expires.
 * Residual: a dynamic `baseURL` (`allowedHosts`) without an explicit
 * `crossSubDomainCookies.domain` resolves the domain per request inside
 * BA; it cannot be known here, so those deployments clear without a
 * `Domain` (set `crossSubDomainCookies.domain` to close it).
 *
 * Also covered: the multi-session plugin's per-session cookies
 * (`<session_token>_multi-<token>`, written with the session token's
 * attributes) and the `__Secure-` / plain counterparts of every name
 * (the same attributes, `Secure` forced on a `__Secure-` name).
 */
export declare function getBetterAuthSessionCookieClearOptions(options: BetterAuthCookieOptions, environment?: SecureCookieEnvironment): (name: string) => CookieOptions | undefined;
/**
 * The base name of a Better Auth cookie chunk, or `undefined` when the
 * name is not a chunk name. BA's chunking session/account stores split
 * an oversized cookie (cookie-cache `session_data`, `account_data`)
 * into `<name>.0` … `<name>.<n>`; the index must be a non-negative
 * safe integer in canonical decimal form, mirroring BA's own
 * `parseCookieChunkIndex`.
 */
export declare function betterAuthCookieChunkBase(name: string): string | undefined;
/**
 * The presented chunks (`<base>.<n>`) of the given cookie names on one
 * request's Cookie header names. Chunk names exist only per request —
 * their count follows the payload size — so no static logout list can
 * hold them; the port enumerates them from the request at logout.
 * Names only: values are never read.
 */
export declare function presentedBetterAuthCookieChunks(presentedNames: Iterable<string>, bases: readonly string[]): readonly string[];

// ===== engine.d.ts =====
import { type IdentityProviderPort, type LoggerPort, type VersionRange } from '@payload-auth-link/core';
import { type BetterAuthOptions, type BetterAuthPlugin } from 'better-auth';
import type { CollectionConfig, Payload } from 'payload';
import { type BetterAuthCollectionsAccess, type BetterAuthCollectionSlugs, type BetterAuthCustomField, type BetterAuthCustomModel, type BetterAuthModelKey, type BetterAuthSchemaMap } from './collections.js';
/** Engine wiring: Payload access plus the builder's Better Auth options. */
export type BetterAuthEngineInput<TOptions extends Omit<BetterAuthOptions, 'database'>> = {
    /** Resolves the Payload instance (cached `getPayload` in Next.js). */
    readonly getPayload: () => Promise<Payload>;
    /** Override any default Better Auth collection slug. */
    readonly slugs?: Partial<BetterAuthCollectionSlugs> | undefined;
    /**
     * Adopt existing collections for these Better Auth models
     * (issue #49): the engine creates no `ba-*` collection for an
     * adopted model and serves it from the host slug instead — the
     * payload-auth layout boots with zero row moves. The host keeps its
     * access, hooks, versions, and admin config; its extra fields are
     * invisible to Better Auth and survive every write (the adapter
     * writes partial updates only).
     */
    readonly schemaMap?: BetterAuthSchemaMap | undefined;
    /**
     * `'payload'` (adopted models only) makes the Payload document id
     * the Better Auth row id, so existing ids never move:
     * `schemaMap: { user: { slug: 'users' } }, idStrategy: 'payload'`.
     * `'ba'` (default) keeps the provider id in a dedicated column on
     * the collections the engine creates.
     */
    readonly idStrategy?: 'ba' | 'payload' | undefined;
    /**
     * Coexistence window (issue #51): legacy payload-auth session
     * cookie names honored READ-ONLY until the ISO `until` date — the
     * lookup re-presents them under the current names, minting always
     * uses the current names, and after the window they are refused.
     * Exposed on the engine as `legacySessionCookieNames` for the
     * cookie helpers and Edge middleware.
     */
    readonly legacySessions?: {
        readonly until: string;
        readonly cookieNames: readonly string[];
    } | undefined;
    /** Custom fields on core models (`additionalFields`, plugin fields). */
    readonly customFields?: Partial<Record<BetterAuthModelKey, readonly BetterAuthCustomField[]>> | undefined;
    /** Custom plugin models (e.g. the organization plugin's tables). */
    readonly customModels?: readonly BetterAuthCustomModel[] | undefined;
    /**
     * Derive `customFields`/`customModels` from Better Auth's own
     * `getAuthTables(options)` (issue #17) and merge them UNDER the
     * builder's explicit declarations — a hand declaration always wins
     * over a derived one, and the derived remainder fills the gaps.
     * Strict hand-declared mode is unchanged when this is absent.
     */
    readonly autoDeclare?: boolean | undefined;
    /** Payload query depth for adapter reads. Defaults to 0. */
    readonly depth?: number | undefined;
    /** Access control for the system collections. Defaults to deny-all. */
    readonly collectionsAccess?: BetterAuthCollectionsAccess | undefined;
    /**
     * Structured logger for wiring-time risky-linking warnings (CAP-12).
     * Defaults to silent (`nullLogger`) — pass the same logger as the
     * plugin to see them. Warnings name the offending
     * `account.accountLinking` setting; they never go to `console.*`.
     */
    readonly logger?: LoggerPort | undefined;
    /**
     * Turn every risky-linking warning into a wiring-time
     * `InvalidConfigError` (CAP-12). Risky means a Better Auth setting
     * that can let an unverified email claim link accounts inside the
     * provider: `account.accountLinking.trustedProviders` naming a
     * provider without `requireEmailVerification`,
     * `allowDifferentEmails: true`, `requireLocalEmailVerified: false`,
     * or `updateUserInfoOnLink: true`. The plugin's verified-email gate
     * still guards Payload linking either way; this option makes the
     * provider-side loosening fail fast instead of merely loud.
     * Defaults to `false` (warn only).
     */
    readonly strictLinking?: boolean | undefined;
    /**
     * Declare that the `twoFactor` plugin's OTP channel
     * (`otpOptions.sendOTP`) is independent of the user's email inbox —
     * for example SMS — so a `/two-factor/verify-otp` code proves a
     * second factor, on the sign-in challenge and on a step-up
     * (GHSA-38wx-69j6-cgm7, GHSA-xgx5-2cj5-x548).
     *
     * Defaults to `false`: the OTP is assumed to reach the same inbox
     * that can reset the password, receive a magic link or an email OTP,
     * or (through the IdP account) back an OAuth sign-in, so an emailed
     * OTP is never MFA proof — a sign-in completed with it is labelled
     * `password`, and an OTP step-up relabels nothing. It also decides
     * whether an OTP-only enrolment counts as an MFA-grade factor whose
     * management (`/two-factor/enable`, `disable`, …) requires an
     * MFA-proven session. TOTP and backup-code verifications are
     * unaffected by this option.
     */
    readonly twoFactorOtpChannelIndependent?: boolean | undefined;
    /**
     * The CAP-7 suspension field on the user row the account-state
     * plugin reads (issue GHSA-xxff-hh6q-c99p). Defaults to
     * `suspendedUntil` — the field the plugin injects. Set it only when
     * a host renamed the field on an adopted users collection.
     */
    readonly suspendedUntilField?: string | undefined;
    /**
     * Keep Better Auth's raw session tokens in its HTTP JSON responses
     * (issues #113 and #116, security-model row 28). Defaults to `false`:
     * the engine's redaction plugin is default-deny — it removes `token`
     * from every session-shaped object in any JSON body (`/get-session`,
     * every `/list-sessions` and `/admin/list-user-sessions` entry, …)
     * and from session-minting and second-factor bodies (`/sign-in/*`,
     * `/sign-up/*`, `/two-factor/verify-*`, …) — HTTP requests only; server-side
     * `auth.api.*` calls keep the full shape. `/revoke-session` then also
     * accepts the listed session `id` in its `token` field. The session
     * cookie stays `HttpOnly`; a JSON copy of it would hand page
     * JavaScript a credential that BA's `bearer()` plugin accepts as
     * `Authorization: Bearer`.
     *
     * Set `true` only for a client that genuinely needs the raw token
     * (e.g. a native app reading it from the sign-in body) — and then
     * install `bearer({ requireSignature: true })` or accept the logged
     * warning. `/multi-session/*` and `/one-time-token/*` hand tokens
     * over by design and are never redacted.
     */
    readonly exposeSessionTokens?: boolean | undefined;
    /**
     * The builder's Better Auth options, passed through verbatim except for
     * `database` (wired to Payload here, so it is omitted from this type),
     * `emailAndPassword` (see below), and
     * `account.accountLinking.updateUserInfoOnLink`, which the engine
     * forces off so a link can never rewrite `email`/`emailVerified`
     * (CAP-12; see `strictLinking`).
     *
     * Framework cookie plugins (`nextCookies()`, `tanstackStartCookies()`,
     * …) belong in `options.plugins`, exactly as Better Auth documents
     * them — the engine never touches the plugin list, so any present or
     * future cookie plugin works without the engine knowing about it.
     */
    readonly options: TOptions;
};
/** Composed engine: instance, port, collections, slugs, cookie names. */
export type BetterAuthEngine<AuthType> = {
    /** The composed Better Auth instance; mount it with the Next adapter. */
    readonly auth: AuthType;
    /** The instance behind `IdentityProviderPort`; register it with the plugin. */
    readonly provider: IdentityProviderPort;
    /** Payload collections for the Better Auth models; spread into the config. */
    readonly collections: CollectionConfig[];
    /** Resolved slugs (defaults plus overrides). */
    readonly slugs: BetterAuthCollectionSlugs;
    /** The adoption map the engine serves (issue #49). */
    readonly schemaMap: BetterAuthSchemaMap;
    /** Legacy cookie names honored during the coexistence window (#51). */
    readonly legacySessionCookieNames: readonly string[];
    /** Session cookie names for middleware fast-paths and logout clearing. */
    readonly sessionCookieNames: readonly string[];
};
/**
 * Compose a Better Auth instance on Payload storage: the Payload database
 * adapter, secure credential defaults, the `IdentityProviderPort`, and
 * the Payload collections.
 *
 * The return type is inferred so Better Auth's own option-driven typing
 * (plugin endpoints, `$Infer` session types) flows through untouched.
 * Builder options (including `plugins`) pass through verbatim — the
 * engine appends nothing, so framework cookie plugins are declared by
 * the builder exactly as Better Auth documents them.
 *
 * Secure defaults: when `emailAndPassword` is present but silent on
 * `requireEmailVerification`, verification is required — builders must
 * explicitly opt out (the example does, and documents why). The user
 * bridge provisions Payload users lazily and idempotently at session
 * verification (see the plugin session service), so no signup
 * ordering can strand a provider user. The engine does compose
 * `databaseHooks` for CAP-14 lifecycle translation (user update /
 * delete, account delete → core `IdentityEvent`s on the port's
 * `onIdentityEvent`), preserving any builder-supplied hooks.
 */

/**
 * A valid API key whose owner (`referenceId`) is not a Better Auth
 * user — in practice an organization-owned key
 * (`apiKey({ references: 'organization' })`), whose `referenceId` is an
 * organization id. `verifyApiKey` resolves a USER; returning an
 * organization id as `userId` would confuse principals, so such keys
 * are refused (fail closed) until organization-owned keys are
 * supported. A `ProviderError`, so every existing failure path treats
 * it as a refusal.
 */
export declare class UnsupportedApiKeyOwnerError extends ProviderError {
    /** The key's owner reference (an organization id, or a deleted user's id). */
    readonly referenceId: string;
    constructor(referenceId: string);
}

    readonly api: {
        readonly getSession: (input: BetterAuthSessionQuery) => Promise<unknown>;

         * entry per presented device session.

            headers: Headers;
        }) => Promise<unknown>) | undefined;
    };
    readonly $context: Promise<{
        readonly internalAdapter: {
            readonly deleteSession: (token: string) => Promise<void>;

             * BA core enumerates one user's live sessions; drives

             */

             * BA core user lookup; present on every real instance. Gates

    }>;

/**
 * The engine's internal session-provenance plugin. Four jobs:
 *
 * 1. **Schema** — one nullable, input-excluded field on Better Auth's
 *    session model: `authenticationMethod`, recording HOW the session
 *    came into being.
 * 2. **Recording** — `hooks.after` on the endpoints whose success
 *    creates (or finally proves) a session: password sign-in records
 *    `password`, the two-factor sign-in verification records
 *    `two-factor` on the session it issues (`password` when the second
 *    factor was an OTP over a channel not declared independent of the
 *    inbox), OAuth callbacks record `oauth`, and so on. The second
 *    factor's own endpoint is the only path that writes `two-factor`,
 *    and server-side mints write `server-mint`/`server-step-up`
 *    (port.ts).
 * 3. **Step-up guard (#110, GHSA-38wx-69j6-cgm7, GHSA-xgx5-2cj5-x548)**
 *    — a `hooks.before` on `/two-factor/verify*` assesses the request
 *    (step-up.ts): the attempt is reserved against the user's failure
 *    budget before any code is checked (refused while locked or once
 *    the budget is spent, shared with the sign-in path; OTP-only users
 *    get a budget too), and a correct code on an EXISTING session
 *    relabels it `two-factor` only when
 *    - the factor was enrolled BEFORE this request (the verification
 *      that enrols a factor proves nothing for the session that
 *      enrolled it; the session keeps its first-factor label),
 *    - no second-factor material was issued or revealed after the
 *      session was created (enrolment, `/two-factor/get-totp-uri`,
 *      `/two-factor/generate-backup-codes` — so a second verification
 *      on the enrolment session is no proof either),
 *    - for `/two-factor/verify-otp`, the OTP channel is declared
 *      independent of the inbox,
 *    - and the signature-verified request cookie names the session BA
 *      resolved for the request (`ctx.context.session`), which is the
 *      session assessed. A failed verification relabels nothing.
 * 4. **Factor-management gate (GHSA-xgx5-2cj5-x548)** — a
 *    `hooks.before` on `/two-factor/enable`, `get-totp-uri`,
 *    `generate-backup-codes` and `disable`: for a user who already
 *    holds an MFA-grade factor, the calling session must itself carry
 *    MFA provenance (`two-factor` / `server-step-up`), else the request
 *    is refused with 403 `TWO_FACTOR_SESSION_REQUIRED` — a password
 *    alone must not read the TOTP secret, mint backup codes, or swap
 *    or remove the factor. Every issuing call marks the factor issued,
 *    so no existing session can be stepped up with what it revealed.
 *
 * The server session API reads this per-session fact for its
 * assurance projection; enrollment flags are never consulted. Paths
 * outside the table and rows that predate the field record nothing —
 * `null` reads as "no recorded method", which the assurance projection
 * treats as proving nothing (fail closed).
 */
export declare function createSessionProvenancePlugin(options?: SessionProvenanceOptions): BetterAuthPlugin;
/** The plugin with its defaults (no independent OTP channel, silent logger). */
export declare const sessionProvenancePlugin: BetterAuthPlugin;

// ===== session-token-redaction.d.ts =====
import { type LoggerPort } from '@payload-auth-link/core';
import type { BetterAuthPlugin } from 'better-auth';
/**
 * Session-token redaction (issue #113, issue #116, security-model row 28).
 *
 * Better Auth's own HTTP endpoints hand the RAW session token to page
 * JavaScript: `/get-session` (`session.token`), `/list-sessions` (the
 * token of EVERY live session of the user), `/admin/list-user-sessions`,
 * the session-minting endpoints' bodies (`/sign-in/email` → `{ token }`,
 * …) and the second-factor verifications (`/two-factor/verify-*` →
 * `{ token, user }`, the presented session's token on a step-up). The
 * session cookie itself is `HttpOnly` precisely so script cannot read
 * it; a JSON copy of the same credential undoes that. With BA's
 * `bearer()` plugin installed (without `requireSignature`), the raw
 * token is a working `Authorization: Bearer` credential — one XSS away
 * from a stolen session on every device.
 *
 * The engine installs this plugin by default (`exposeSessionTokens`
 * opts out). For HTTP requests only — server-side `auth.api.*` calls
 * run in trusted code and keep the full shape (the port's own
 * `getSession` reads `session.token`) — its after-hook is default-deny:
 * it removes `token` from
 *
 * - every session-shaped object anywhere in the JSON body (a record
 *   carrying a string `token` next to a session field — `userId`,
 *   `expiresAt`, `ipAddress`, `userAgent` — or found under a `session`
 *   key),
 * - every `/list-sessions` entry (each keeps its `id`),
 * - a top-level body that pairs a string `token` with a `user` record
 *   (the session-minting and second-factor bodies),
 * - and any top-level `token` EQUAL to the session token the response
 *   just minted (`ctx.context.newSession`).
 *
 * A value that merely shares the key name (the `jwt` plugin's `/token`,
 * a one-time token, …) is never touched. Token hand-off protocols are
 * left alone by design: `/multi-session/*` (switching and revoking
 * device sessions takes the token) and `/one-time-token/*` (the
 * exchange returns the session).
 *
 * `/revoke-session` takes a session token; with the tokens redacted
 * from `/list-sessions`, its before-hook also accepts the listed
 * session `id` (the same opaque handle the server API uses, issue
 * #80) and resolves it to the token server-side. BA's endpoint still
 * enforces that the session belongs to the caller.
 */
export declare const sessionTokenRedactionPluginId = "payload-auth-link-session-token-redaction";
/**
 * The redacted body for one HTTP response, or `undefined` when nothing
 * needs to change. Pure — the hook below feeds it the endpoint path,
 * the endpoint's returned value, and the session the response minted.
 * Exported for the unit suite.
 */
export declare function redactSessionTokens(path: string | undefined, returned: unknown, newSession: unknown): unknown;
/** The hook-context fields this plugin reads (Better Auth 1.7). */
export type RedactionHookContext = {
    readonly path?: string | undefined;
    readonly request?: unknown;
    readonly body?: unknown;
    readonly context: {
        readonly returned?: unknown;
        readonly newSession?: unknown;
        readonly adapter?: {
            readonly findOne?: ((input: {
                model: string;
                where: {
                    field: string;
                    value: string;
                }[];
            }) => Promise<unknown>) | undefined;
        };
    };
};
/**
 * `/revoke-session` body rewrite: a `token` that names a session ROW id
 * becomes that row's token. A real token never matches a row id, so
 * it passes through unchanged; an unknown value too (BA then answers
 * as it always does). Exported for the unit suite.
 */
export declare function resolveRevokeHandle(ctx: RedactionHookContext): Promise<{
    context: {
        body: Record<string, unknown>;
    };
} | undefined>;
/** The engine's session-token redaction plugin (see the module doc). */
export declare const sessionTokenRedactionPlugin: BetterAuthPlugin;

type InstalledPlugin = {

// ===== step-up.d.ts =====
import type { LoggerPort } from '@payload-auth-link/core';
/** Better Auth's `twoFactor({ accountLockout })`, resolved with BA's own defaults. */
export type LockoutConfig = {
    readonly enabled: boolean;
    readonly maxFailedAttempts: number;
    readonly durationMs: number;
};
/**
 * A reserved slot in the user's failure budget.
 *
 * - `two-factor-row`: Better Auth's own counter on the user's
 *   `twoFactor` row; `reserved` is the count this attempt brought it to.
 * - `attempt-rows`: one `verification` row per counted attempt, for a
 *   user without a `twoFactor` row (OTP-only enrolment); `attemptId` is
 *   this attempt's row.
 */
export type FailureBudget = {
    readonly kind: 'two-factor-row';
    readonly twoFactorId: string;
    readonly reserved: number;
} | {
    readonly kind: 'attempt-rows';
    readonly identifier: string;
    readonly attemptId: string;
};
/**
 * What the before-hook established about one second-factor request.
 *
 * - `sign-in`: no session was presented; Better Auth's challenge path
 *   runs (and counts users with a `twoFactor` row itself). `provesMfa`
 *   is false for `/two-factor/verify-otp` unless the OTP channel is
 *   declared independent of the inbox: an emailed OTP after a password
 *   the same inbox can reset is not a second factor.
 * - `step-up`: a verification on the existing session `token`.
 *   `provesMfa` is true only when the factor was enrolled before this
 *   request (`enrolment` is false), no second-factor material was
 *   issued or revealed after the session was created, and, for the OTP
 *   route, the OTP channel is declared independent.
 *
 * `budget` is the attempt this request reserved (`null`: none — lockout
 * disabled, or Better Auth counts this attempt itself).
 */
export type StepUpAssessment = {
    readonly kind: 'sign-in';
    readonly userId: string | null;
    readonly provesMfa: boolean;
    readonly budget: FailureBudget | null;
    readonly lockout: LockoutConfig;
} | {
    readonly kind: 'step-up';
    readonly token: string;
    readonly userId: string;
    readonly provesMfa: boolean;
    readonly enrolment: boolean;
    readonly budget: FailureBudget | null;
    readonly lockout: LockoutConfig;
};
/**
 * What the before-hook established about a request that manages the
 * user's second factor (`/two-factor/enable`, `get-totp-uri`,
 * `generate-backup-codes`, `disable`).
 */
export type FactorManagementAssessment = {
    readonly userId: string;
    /** The presented session's recorded method (`null`: none recorded). */
    readonly method: string | null;
    /**
     * The user already holds a factor that proves MFA: two-factor is on
     * AND a confirmed TOTP row exists or the OTP channel is declared
     * independent. Only then can the factor be stolen by revealing or
     * rotating it, so only then is an MFA-proven session required.
     */
    readonly holdsMfaFactor: boolean;
};
/** Runs `task` after every earlier task queued under the same key. */
export type Serializer = <T>(key: string, task: () => Promise<T>) => Promise<T>;
/**
 * A per-key FIFO queue (in-process). Attempt reservations and budget
 * writes for one user run one at a time, so Better Auth's
 * compare-and-swap counter increment never contends with itself.
 */
export declare function createSerializer(): Serializer;
/** Wiring for the step-up guard. */
export type StepUpOptions = {
    /** See `BetterAuthEngineInput.twoFactorOtpChannelIndependent`. */
    readonly otpChannelIndependent: boolean;
    readonly logger: LoggerPort;
    /** Serializes reservations and budget writes per user. */
    readonly serialize: Serializer;
};
/** One Better Auth adapter call, as the guard sees it (query in, row(s) out). */
export type AdapterOperation = (query: Record<string, unknown>) => Promise<unknown>;
/** The Better Auth 1.7 hook-context fields this module reads. */
export type StepUpHookContext = {
    readonly path?: string | undefined;
    readonly getSignedCookie?: ((key: string, secret: string) => Promise<string | null | false>) | undefined;
    readonly context: {
        session?: unknown;
        readonly secret?: unknown;
        readonly getPlugin?: ((id: string) => unknown) | undefined;
        readonly createAuthCookie?: ((name: string) => unknown) | undefined;
        readonly internalAdapter?: {
            readonly findSession?: ((token: string) => Promise<unknown>) | undefined;
            readonly findUserById?: ((id: string) => Promise<unknown>) | undefined;
            readonly findVerificationValue?: ((identifier: string) => Promise<unknown>) | undefined;
        };
        readonly adapter?: {
            readonly findOne?: AdapterOperation | undefined;
            readonly findMany?: AdapterOperation | undefined;
            readonly count?: AdapterOperation | undefined;
            readonly create?: AdapterOperation | undefined;
            readonly incrementOne?: AdapterOperation | undefined;
            readonly update?: AdapterOperation | undefined;
            readonly delete?: AdapterOperation | undefined;
            readonly deleteMany?: AdapterOperation | undefined;
        };
    };
};
/** BA's `resolveAccountLockoutConfig`, read from the installed `two-factor` plugin. */
export declare function lockoutConfigOf(ctx: StepUpHookContext): LockoutConfig;
/**
 * When the user's second-factor material was last issued or revealed
 * (epoch ms), or `null` when no marker exists (an enrolment that
 * predates this guard). An unreadable marker counts as newer than any
 * session (fail closed).
 */
export declare function factorIssuedAt(ctx: StepUpHookContext, userId: string): Promise<number | null>;
/**
 * Record that the user's second-factor material was issued or revealed
 * now (GHSA-xgx5-2cj5-x548): no session created before this moment can
 * be relabelled `two-factor` by a step-up afterwards. The new marker is
 * written before the old ones are removed, so a concurrent reader
 * always sees one.
 */
export declare function markFactorIssued(ctx: StepUpHookContext, userId: string): Promise<void>;
/**
 * Assess one second-factor request BEFORE its endpoint runs, and
 * reserve its attempt. For a step-up, the state that decides it is
 * read FRESH from storage, never from the cookie cache: the session
 * row (its creation time), the user's enrolment flag, the `twoFactor`
 * row and the issuance marker.
 *
 * Throws Better Auth's own `ACCOUNT_TEMPORARILY_LOCKED` (429) while the
 * account is locked or once the budget is spent — the request is
 * refused before any code is checked, exactly like the sign-in path.
 * A verification that would ENROL a factor marks the factor issued
 * first, so no session that exists while it completes can later be
 * relabelled by it.
 */
export declare function assessSecondFactorRequest(ctx: StepUpHookContext, resolveSession: () => Promise<unknown>, options: StepUpOptions): Promise<StepUpAssessment>;
/**
 * Assess a request that manages the user's second factor, BEFORE its
 * endpoint runs. Returns `null` when no session is presented (Better
 * Auth's session middleware then refuses the request itself).
 */
export declare function assessFactorManagement(ctx: StepUpHookContext, resolveSession: () => Promise<unknown>, options: Pick<StepUpOptions, 'otpChannelIndependent'>): Promise<FactorManagementAssessment | null>;
/** How a second-factor endpoint ended, as far as the counter cares. */
export type SecondFactorOutcome = 'success' | 'failed-guess' | 'failed';
/**
 * Classify the endpoint's result: a 200 body is `success`; Better
 * Auth's own wrong-code errors are `failed-guess` (they spend the
 * budget); anything else (expired OTP, too many attempts on one code,
 * a server error) is `failed` and neither spends nor resets it.
 */
export declare function outcomeOf(returned: unknown): SecondFactorOutcome;
/**
 * Settle the attempt the before-hook reserved: a success resets the
 * budget (Better Auth counts consecutive failures only), a wrong guess
 * keeps its reservation and locks the row once the budget is spent,
 * and any other failure releases the reservation. A write that fails
 * is logged and swallowed — the request's own outcome stands.
 */
export declare function settleSecondFactor(ctx: StepUpHookContext, assessment: StepUpAssessment, outcome: SecondFactorOutcome, options: Pick<StepUpOptions, 'logger' | 'serialize'>): Promise<void>;
